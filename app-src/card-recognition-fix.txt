

  // Robust Scryfall card-name recognition and saved-table repair.
  // Handles common typos/diacritics, then falls back to a scored Scryfall
  // name search so real cards do not silently become blank fallback objects.
  (() => {
    const SCRYFALL_SEARCH = 'https://api.scryfall.com/cards/search';
    const forgeBaseFetchScryfallCards = fetchScryfallCards;
    const forgeBaseNormalizeLoadedState = normalizeLoadedState;
    const forgeBaseMoveCommanderOutOfLibrary = moveCommanderOutOfLibrary;
    let forgeRepairPromise = null;

    function forgeCardLookupKey(value) {
      return String(value || '')
        .normalize('NFD')
        .replace(/[\u0300-\u036f]/g, '')
        .replace(/[Ææ]/g, m => m === 'Æ' ? 'AE' : 'ae')
        .replace(/[Œœ]/g, m => m === 'Œ' ? 'OE' : 'oe')
        .replace(/[’‘`]/g, "'")
        .replace(/[‐‑‒–—]/g, '-')
        .toLocaleLowerCase()
        .replace(/[^a-z0-9]+/g, ' ')
        .trim()
        .replace(/\s+/g, ' ');
    }

    const forgeCardAliases = new Map([
      ['calvary pegasus', 'Cavalry Pegasus'],
      ['eomer king of rohan', 'Éomer, King of Rohan']
    ]);

    function forgeAliasForCardName(name) {
      return forgeCardAliases.get(forgeCardLookupKey(name)) || String(name || '').trim();
    }

    function forgeLevenshtein(a, b) {
      const left = forgeCardLookupKey(a);
      const right = forgeCardLookupKey(b);
      if (left === right) return 0;
      if (!left) return right.length;
      if (!right) return left.length;
      const prev = Array.from({length:right.length + 1}, (_,i) => i);
      const curr = new Array(right.length + 1);
      for (let i=1;i<=left.length;i++) {
        curr[0] = i;
        for (let j=1;j<=right.length;j++) {
          curr[j] = Math.min(
            curr[j-1] + 1,
            prev[j] + 1,
            prev[j-1] + (left[i-1] === right[j-1] ? 0 : 1)
          );
        }
        for (let j=0;j<=right.length;j++) prev[j] = curr[j];
      }
      return prev[right.length];
    }

    function forgeFindResolvedData(map, name) {
      if (!map || !name) return null;
      const lower = String(name).toLocaleLowerCase();
      const direct = map.get(lower);
      if (direct) return direct;
      const wantedKey = forgeCardLookupKey(name);
      for (const item of new Set(map.values())) {
        if (item?.name && forgeCardLookupKey(item.name) === wantedKey) return item;
      }
      return null;
    }

    async function forgeNamedFuzzy(name) {
      if (!name) return null;
      try {
        const res = await fetch(`${SCRYFALL_NAMED}?fuzzy=${encodeURIComponent(name)}`);
        if (!res.ok) return null;
        return await res.json();
      } catch {
        return null;
      }
    }

    async function forgeSearchByNameWord(wanted) {
      const wantedKey = forgeCardLookupKey(wanted);
      const words = [...new Set(wantedKey.split(' ').filter(word => word.length >= 4))]
        .sort((a,b) => b.length - a.length);
      let best = null;

      for (const word of words) {
        try {
          const query = `name:${word}`;
          const res = await fetch(`${SCRYFALL_SEARCH}?q=${encodeURIComponent(query)}&unique=cards&order=name`);
          if (!res.ok) continue;
          const payload = await res.json();
          for (const item of payload.data || []) {
            if (!item?.name) continue;
            const distance = forgeLevenshtein(wanted, item.name);
            const maxLen = Math.max(wantedKey.length, forgeCardLookupKey(item.name).length, 1);
            const ratio = distance / maxLen;
            if (!best || ratio < best.ratio || (ratio === best.ratio && distance < best.distance)) {
              best = { item, distance, ratio };
            }
          }
        } catch {}

        if (best && (best.distance <= 2 || best.ratio <= 0.16)) break;
      }

      if (!best) return null;
      const maxAllowed = Math.max(2, Math.floor(Math.max(1,wantedKey.length) * 0.28));
      return best.distance <= maxAllowed || best.ratio <= 0.28 ? best.item : null;
    }

    async function forgeResolveOneCardName(wanted) {
      const alias = forgeAliasForCardName(wanted);
      const attempts = [...new Set([alias, String(wanted || '').trim()].filter(Boolean))];
      for (const attempt of attempts) {
        const item = await forgeNamedFuzzy(attempt);
        if (item) return item;
      }
      return forgeSearchByNameWord(wanted);
    }

    fetchScryfallCards = async function(names, statusEl=null) {
      const requested = [...new Map((names || []).filter(Boolean).map(name => [String(name).toLocaleLowerCase(), String(name)])).values()];
      const canonicalRequests = requested.map(forgeAliasForCardName);
      const baseMap = await forgeBaseFetchScryfallCards(canonicalRequests, statusEl);
      const results = new Map(baseMap);
      const unresolved = [];

      for (const wanted of requested) {
        const alias = forgeAliasForCardName(wanted);
        const item = forgeFindResolvedData(baseMap, alias) || forgeFindResolvedData(baseMap, wanted);
        if (item) {
          results.set(wanted.toLocaleLowerCase(), item);
          results.set(item.name.toLocaleLowerCase(), item);
        } else {
          unresolved.push(wanted);
        }
      }

      for (const wanted of unresolved) {
        const item = await forgeResolveOneCardName(wanted);
        if (!item) continue;
        results.set(wanted.toLocaleLowerCase(), item);
        results.set(item.name.toLocaleLowerCase(), item);
        await new Promise(resolve => setTimeout(resolve, 55));
      }

      if (statusEl) {
        const resolvedCount = requested.filter(name => results.has(name.toLocaleLowerCase())).length;
        setStatus(
          statusEl,
          resolvedCount === requested.length
            ? `Loaded ${resolvedCount} unique cards.`
            : `Loaded ${resolvedCount}/${requested.length} unique cards.`,
          resolvedCount ? 'success' : 'error'
        );
      }
      return results;
    };

    moveCommanderOutOfLibrary = function(name, doCheckpoint=true) {
      const wanted = forgeCardLookupKey(forgeAliasForCardName(name));
      const id = state.zones.library.find(id => {
        const c = card(id);
        if (!c?.name) return false;
        const full = forgeCardLookupKey(c.name);
        const front = forgeCardLookupKey(String(c.name).split(' // ')[0]);
        return full === wanted || front === wanted;
      });
      if (!id) return forgeBaseMoveCommanderOutOfLibrary(name, doCheckpoint);
      if (doCheckpoint) checkpoint();
      removeFromZones(id);
      state.zones.command.push(id);
      state.commanderName = card(id)?.name || name;
      return true;
    };

    async function forgeRepairMissingRecognizedCards() {
      if (forgeRepairPromise) return forgeRepairPromise;
      forgeRepairPromise = (async () => {
        const missing = Object.values(state.cards || {}).filter(c => {
          if (!c || c.custom || !c.name || c.name === 'Unknown card') return false;
          const hasFaceImage = Array.isArray(c.faces) && c.faces.some(face => face?.image);
          return !c.image && !hasFaceImage;
        });
        if (!missing.length) return 0;

        const names = [...new Map(missing.map(c => [c.name.toLocaleLowerCase(), c.name])).values()];
        const map = await fetchScryfallCards(names, null);
        let repaired = 0;

        for (const c of missing) {
          const oldName = c.name;
          const data = map.get(oldName.toLocaleLowerCase()) || forgeFindResolvedData(map, forgeAliasForCardName(oldName));
          if (!data) continue;
          Object.assign(c, cardTemplate(data, oldName));
          if (state.commanderName && forgeCardLookupKey(state.commanderName) === forgeCardLookupKey(oldName)) {
            state.commanderName = c.name;
          }
          repaired++;
        }

        if (repaired) {
          render();
          scheduleSave();
          toast(`Recognized ${repaired} missing card${repaired === 1 ? '' : 's'} from Scryfall.`);
        }
        return repaired;
      })().finally(() => { forgeRepairPromise = null; });
      return forgeRepairPromise;
    }

    normalizeLoadedState = function(raw) {
      const next = forgeBaseNormalizeLoadedState(raw);
      setTimeout(() => {
        if (state === next && next.started) forgeRepairMissingRecognizedCards();
      }, 0);
      return next;
    };
  })();

  // Visual picker for Add any card.
  // Search first, then explicitly select the exact card instead of trusting a
  // single fuzzy result.
  (() => {
    const AUTOCOMPLETE_URL = 'https://api.scryfall.com/cards/autocomplete';
    const modal = document.querySelector('#add-card-modal .small-modal');
    const input = document.getElementById('add-card-name');
    const status = document.getElementById('add-card-status');
    const submit = document.getElementById('add-card-submit');
    const zoneSelect = document.getElementById('add-card-zone');
    if (!modal || !input || !status || !submit || !zoneSelect) return;

    let selectedCard = null;
    let searchResults = [];
    let searchTimer = null;
    let searchSequence = 0;

    if (!document.getElementById('forge-add-card-picker-style')) {
      const style = document.createElement('style');
      style.id = 'forge-add-card-picker-style';
      style.textContent = `
        #add-card-modal .small-modal{
          width:min(980px,96vw)!important;
          max-height:94vh!important;
        }
        #forge-card-search-help{
          margin:-2px 0 8px;
          font-size:11px;
          color:#9fb0aa;
        }
        #forge-card-search-results{
          display:grid;
          grid-template-columns:repeat(auto-fill,minmax(138px,1fr));
          gap:12px;
          max-height:50vh;
          overflow:auto;
          padding:10px;
          margin:8px 0 14px;
          border:1px solid rgba(91,116,107,.58);
          border-radius:14px;
          background:rgba(6,12,10,.38);
        }
        #forge-card-search-results:empty{display:none}
        .forge-card-search-result{
          display:flex;
          flex-direction:column;
          gap:7px;
          min-width:0;
          padding:7px;
          border:1px solid rgba(91,116,107,.55);
          border-radius:12px;
          background:rgba(18,29,25,.96);
          color:#eef4f0;
          text-align:left;
          cursor:pointer;
          transition:transform .12s ease,border-color .12s ease,box-shadow .12s ease,background .12s ease;
        }
        .forge-card-search-result:hover,
        .forge-card-search-result:focus-visible{
          transform:translateY(-2px);
          border-color:rgba(241,199,121,.88);
          box-shadow:0 10px 26px rgba(0,0,0,.42);
          outline:none;
        }
        .forge-card-search-result.selected{
          border-color:#f1c779;
          box-shadow:0 0 0 2px rgba(241,199,121,.22),0 10px 28px rgba(0,0,0,.48);
          background:rgba(31,43,36,.99);
        }
        .forge-card-search-result img,
        .forge-card-search-fallback{
          display:block;
          width:100%;
          aspect-ratio:488/680;
          border-radius:9px;
          object-fit:cover;
          background:#17211f;
        }
        .forge-card-search-fallback{
          padding:10px;
          font-size:11px;
          font-weight:800;
        }
        .forge-card-search-result strong{
          font-size:11px;
          line-height:1.25;
        }
        .forge-card-search-result small{
          color:#9fb0aa;
          font-size:9px;
          line-height:1.25;
        }
        .forge-card-search-message{
          grid-column:1/-1;
          margin:0;
          padding:18px;
          text-align:center;
          color:#9fb0aa;
          font-size:12px;
        }
        @media(max-width:640px){
          #forge-card-search-results{grid-template-columns:repeat(2,minmax(0,1fr));gap:8px;padding:8px}
        }
      `;
      document.head.appendChild(style);
    }

    const help = document.createElement('div');
    help.id = 'forge-card-search-help';
    help.textContent = 'Type at least 2 characters, then choose the exact card from the results.';
    const resultsRoot = document.createElement('div');
    resultsRoot.id = 'forge-card-search-results';
    resultsRoot.setAttribute('role','listbox');
    resultsRoot.setAttribute('aria-label','Matching Magic cards');
    input.closest('label')?.insertAdjacentElement('afterend', help);
    help.insertAdjacentElement('afterend', resultsRoot);

    submit.textContent = 'Add selected card';
    submit.disabled = true;

    function pickerImage(data) {
      if (!data) return '';
      if (data.image_uris) return data.image_uris.small || data.image_uris.normal || data.image_uris.large || '';
      const face = data.card_faces?.[0];
      return face?.image_uris?.small || face?.image_uris?.normal || face?.image_uris?.large || '';
    }

    function resetPicker(clearInput=false) {
      selectedCard = null;
      searchResults = [];
      searchSequence++;
      clearTimeout(searchTimer);
      submit.disabled = true;
      resultsRoot.innerHTML = '';
      if (clearInput) input.value = '';
    }

    function renderResults() {
      resultsRoot.innerHTML = '';
      if (!searchResults.length) {
        const empty = document.createElement('p');
        empty.className = 'forge-card-search-message';
        empty.textContent = 'No matching cards found.';
        resultsRoot.appendChild(empty);
        return;
      }

      for (const data of searchResults) {
        const button = document.createElement('button');
        button.type = 'button';
        button.className = 'forge-card-search-result';
        button.dataset.scryfallId = data.id || '';
        button.setAttribute('role','option');
        button.setAttribute('aria-selected', selectedCard?.id === data.id ? 'true' : 'false');
        if (selectedCard?.id === data.id) button.classList.add('selected');

        const image = pickerImage(data);
        if (image) {
          const img = document.createElement('img');
          img.src = image;
          img.alt = data.name || 'Magic card';
          img.loading = 'lazy';
          img.draggable = false;
          button.appendChild(img);
        } else {
          const fallback = document.createElement('div');
          fallback.className = 'forge-card-search-fallback';
          fallback.textContent = data.name || 'Card';
          button.appendChild(fallback);
        }

        const name = document.createElement('strong');
        name.textContent = data.name || 'Unnamed card';
        button.appendChild(name);

        const type = document.createElement('small');
        type.textContent = data.type_line || data.card_faces?.[0]?.type_line || '';
        button.appendChild(type);

        button.addEventListener('click', () => {
          selectedCard = data;
          input.value = data.name || input.value;
          submit.disabled = false;
          setStatus(status, `Selected ${data.name}. Choose a zone, then add it.`, 'success');
          renderResults();
        });
        resultsRoot.appendChild(button);
      }
    }

    async function runSearch(rawQuery) {
      const query = String(rawQuery || '').trim();
      const seq = ++searchSequence;
      selectedCard = null;
      submit.disabled = true;

      if (query.length < 2) {
        resultsRoot.innerHTML = '';
        setStatus(status, 'Type at least 2 characters to search.');
        return;
      }

      resultsRoot.innerHTML = '<p class="forge-card-search-message">Searching Scryfall…</p>';
      setStatus(status, 'Searching for matching cards…');

      try {
        const autoRes = await fetch(`${AUTOCOMPLETE_URL}?q=${encodeURIComponent(query)}&include_extras=true`);
        if (!autoRes.ok) throw new Error(`Search failed (${autoRes.status})`);
        const autoPayload = await autoRes.json();
        if (seq !== searchSequence) return;

        const names = (autoPayload.data || []).slice(0, 16);
        if (!names.length) {
          searchResults = [];
          renderResults();
          setStatus(status, 'No matching card names found.', 'error');
          return;
        }

        const map = await fetchScryfallCards(names, null);
        if (seq !== searchSequence) return;
        searchResults = names
          .map(name => map.get(String(name).toLocaleLowerCase()))
          .filter(Boolean)
          .filter((item,index,array) => array.findIndex(other => other.id === item.id) === index);

        renderResults();
        setStatus(status, `${searchResults.length} matching card${searchResults.length === 1 ? '' : 's'} found. Select one.`);
      } catch (err) {
        if (seq !== searchSequence) return;
        searchResults = [];
        resultsRoot.innerHTML = '<p class="forge-card-search-message">Search unavailable. Try again.</p>';
        setStatus(status, err?.message || 'Could not search Scryfall.', 'error');
      }
    }

    input.addEventListener('input', () => {
      selectedCard = null;
      submit.disabled = true;
      clearTimeout(searchTimer);
      const query = input.value;
      searchTimer = setTimeout(() => runSearch(query), 220);
    });

    input.addEventListener('focus', () => {
      if (input.value.trim().length >= 2 && !resultsRoot.children.length) runSearch(input.value);
    });

    // Reset old results whenever the user intentionally opens Add any card.
    document.getElementById('add-card-btn')?.addEventListener('click', () => {
      resetPicker(false);
      setStatus(status, 'Start typing to search for a card.');
    });
    document.getElementById('tool-add-card')?.addEventListener('click', () => {
      resetPicker(false);
      setStatus(status, 'Start typing to search for a card.');
    });

    addAnyCard = async function() {
      if (!selectedCard) {
        setStatus(status, 'Select a card from the search results first.', 'error');
        if (input.value.trim().length >= 2) runSearch(input.value);
        return;
      }

      submit.disabled = true;
      try {
        const instance = makeInstance(cardTemplate(selectedCard, selectedCard.name), {outsideDeck:true});
        let target = zoneSelect.value;
        let placement = 'bottom';
        if (target === 'library-top') { target = 'library'; placement = 'top'; }
        if (target === 'library-bottom') { target = 'library'; placement = 'bottom'; }

        checkpoint();
        addInstanceToState(instance, target, placement);
        if (target === 'battlefield') { instance.x = 40; instance.y = 40; }
        logEvent(`Added ${instance.name} to ${target}.`);
        render();
        scheduleSave();
        closeModal('add-card-modal');
        toast(`${instance.name} added.`);
        resetPicker(true);
        setStatus(status, '');
      } finally {
        submit.disabled = !selectedCard;
      }
    };
  })();
