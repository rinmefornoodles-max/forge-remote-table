
  // Keep the selected D20 value on a face that points directly toward the player.
  // The visual renderer still handles perspective/shading; this file only supplies
  // the correct physical orientation for whichever value is currently selected.
  (() => {
    const PHI = (1 + Math.sqrt(5)) / 2;
    const VERTICES = [
      [-1, PHI, 0], [1, PHI, 0], [-1, -PHI, 0], [1, -PHI, 0],
      [0, -1, PHI], [0, 1, PHI], [0, -1, -PHI], [0, 1, -PHI],
      [PHI, 0, -1], [PHI, 0, 1], [-PHI, 0, -1], [-PHI, 0, 1]
    ];
    const FACES = [
      [0,11,5],[0,5,1],[0,1,7],[0,7,10],[0,10,11],
      [1,5,9],[5,11,4],[11,10,2],[10,7,6],[7,1,8],
      [3,9,4],[3,4,2],[3,2,6],[3,6,8],[3,8,9],
      [4,9,5],[2,4,11],[6,2,10],[8,6,7],[9,8,1]
    ];
    const lastValueById = new Map();

    const sub = (a,b) => [a[0]-b[0],a[1]-b[1],a[2]-b[2]];
    const cross = (a,b) => [
      a[1]*b[2]-a[2]*b[1],
      a[2]*b[0]-a[0]*b[2],
      a[0]*b[1]-a[1]*b[0]
    ];
    const dot = (a,b) => a[0]*b[0]+a[1]*b[1]+a[2]*b[2];
    const normalize = v => {
      const length = Math.hypot(v[0],v[1],v[2]) || 1;
      return [v[0]/length,v[1]/length,v[2]/length];
    };
    const deg = radians => radians * 180 / Math.PI;

    // The app rotates X, then Y, then Z. These angles make a face normal become
    // +Z (straight toward the viewer). Z can then be used only as an in-face twist.
    const FACE_FORWARD_ANGLES = FACES.map(face => {
      const a=VERTICES[face[0]], b=VERTICES[face[1]], c=VERTICES[face[2]];
      const centroid=[(a[0]+b[0]+c[0])/3,(a[1]+b[1]+c[1])/3,(a[2]+b[2]+c[2])/3];
      let normal=cross(sub(b,a),sub(c,a));
      if (dot(normal,centroid) < 0) normal=normal.map(v=>-v);
      normal=normalize(normal);
      const rx=Math.atan2(normal[1],normal[2]);
      const yz=Math.hypot(normal[1],normal[2]);
      const ry=Math.atan2(-normal[0],yz);
      return [deg(rx),deg(ry)];
    });

    function faceForwardDie(die) {
      if (!die || die.kind !== 'd20') return false;
      const value=Math.max(1,Math.min(20,Number(die.value)||1));
      const [rx,ry]=FACE_FORWARD_ANGLES[(value-1)%20];
      let changed=false;
      if (Math.abs((Number(die.rx)||0)-rx) > .01) { die.rx=rx; changed=true; }
      if (Math.abs((Number(die.ry)||0)-ry) > .01) { die.ry=ry; changed=true; }

      // Whenever the selected value changes, remove any old sideways twist so
      // the chosen face presents cleanly. A later explicit Rotate 45° action is
      // preserved because the value itself has not changed.
      const previous=lastValueById.get(die.id);
      if (previous !== value) {
        lastValueById.set(die.id,value);
        if (Math.abs(Number(die.rz)||0) > .01) { die.rz=0; changed=true; }
      }
      return changed;
    }

    function faceForwardAll() {
      if (!Array.isArray(state?.dice)) return false;
      let changed=false;
      for (const die of state.dice) changed=faceForwardDie(die)||changed;
      return changed;
    }

    function nudgeRendererIfNeeded() {
      if (!faceForwardAll()) return;
      // The real-D20 renderer watches style mutations. Touching a harmless CSS
      // custom property guarantees it gets one render using the corrected angles.
      document.querySelectorAll('#battlefield .forge-table-die.forge-die-d20').forEach(el => {
        el.style.setProperty('--forge-face-forward',String(Date.now()));
      });
    }

    // Apply before the first animation-frame paint from real-d20.txt.
    faceForwardAll();

    const battlefield=document.getElementById('battlefield');
    if (battlefield) {
      new MutationObserver(() => nudgeRendererIfNeeded()).observe(battlefield, {
        childList:true,subtree:true,attributes:true,attributeFilter:['style']
      });
    }

    // Menu value changes rebuild the die. Run once more after that event finishes
    // so the selected value is guaranteed to be the face aimed at the player.
    document.addEventListener('click',e => {
      if (!e.target.closest?.('[data-die-menu-value]')) return;
      setTimeout(nudgeRendererIfNeeded,0);
    },true);

    requestAnimationFrame(nudgeRendererIfNeeded);
  })();
