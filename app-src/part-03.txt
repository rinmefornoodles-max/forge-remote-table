  function toggleTapped(id) {

  }

  function inspectorAction(action) {
    const c=card(selectedId); if(!c) return;
    if(action==='tap') { checkpoint(); c.tapped=!c.tapped; }
    else if(action==='facedown') { checkpoint(); c.faceDown=!c.faceDown; }
    else if(action==='transform' && c.faces?.length>1) { checkpoint(); c.faceIndex=(c.faceIndex+1)%c.faces.length; }
    else if(action==='plus') { checkpoint(); c.counters['+1/+1']=(Number(c.counters['+1/+1'])||0)+1; }
    else if(action==='minus') { checkpoint(); c.counters['-1/-1']=(Number(c.counters['-1/-1'])||0)+1; }
    else if(action==='counter') {
      const name=prompt('Counter name:', 'counter'); if(!name) return;
      const current=Number(c.counters[name])||0; const value=prompt(`Set “${name}” counters:`,String(current+1)); if(value===null) return;
      const n=Number(value); if(!Number.isFinite(n)) return; checkpoint(); c.counters[name]=Math.trunc(n);
    } else if(action==='copy') {
      checkpoint(); const clone=makeInstance(deepCopy(c),{id:uid(),token:true,x:(Number(c.x)||25)+28,y:(Number(c.y)||25)+28,tapped:false});
      addInstanceToState(clone,'battlefield'); logEvent(`Created a copy of ${c.name}.`);
    } else if(action==='note') {
      const value=prompt('Note for this card:',c.note||''); if(value===null)return; checkpoint(); c.note=value;
    } else if(action==='delete') {
      if(!confirm(`Delete ${c.name} from this table? You can Undo afterward.`)) return;
      checkpoint(); removeFromZones(c.id); delete state.cards[c.id]; selectedId=null; closeModal('inspector');
    }
    render(); scheduleSave(); if(selectedId && card(selectedId)) openInspector(selectedId);
  }

  async function addAnyCard() {
    const name=$('#add-card-name').value.trim(); const status=$('#add-card-status');
    if(!name) return setStatus(status,'Enter a card name.','error');
    setStatus(status,'Searching Scryfall…'); $('#add-card-submit').disabled=true;
    try{
      const res=await fetch(`${SCRYFALL_NAMED}?fuzzy=${encodeURIComponent(name)}`);
      if(!res.ok) throw new Error('Card not found');
      const data=await res.json();
      const instance=makeInstance(cardTemplate(data,name),{outsideDeck:true});
      let target=$('#add-card-zone').value; let placement='bottom';
      if(target==='library-top'){target='library';placement='top';}
      if(target==='library-bottom'){target='library';placement='bottom';}
      checkpoint(); addInstanceToState(instance,target,placement);
      if(target==='battlefield'){instance.x=40;instance.y=40;}
      logEvent(`Added ${instance.name} to ${target}.`); render(); scheduleSave(); closeModal('add-card-modal'); toast(`${instance.name} added.`);
    }catch(err){setStatus(status,err.message||'Card not found.','error');}
    finally{$('#add-card-submit').disabled=false;}
  }

  function newGameSameDeck() {
    if(!confirm('Start a fresh game with this same deck? Tokens/custom added cards are removed. You can Undo once afterward.')) return;
    checkpoint();
    const commanderIds = [];
    const all=[];
    for(const [id,c] of Object.entries(state.cards)) {
      if(c.token || c.custom || c.outsideDeck) { delete state.cards[id]; continue; }
      c.tapped=false;c.faceDown=false;c.counters={};c.x=null;c.y=null;c.note='';c.faceIndex=0;
      if(state.commanderName && c.name.toLocaleLowerCase()===state.commanderName.toLocaleLowerCase() && !commanderIds.length) commanderIds.push(id); else all.push(id);
    }
    state.zones={library:all,hand:[],battlefield:[],graveyard:[],exile:[],command:commanderIds};
    state.life=40;state.poison=0;state.energy=0;state.commanderTax=0;state.turn=1;state.damage=state.damage.map(d=>({...d,value:0}));
    const a=state.zones.library;for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];}
    for(let i=0;i<7&&a.length;i++)state.zones.hand.push(a.shift());
    logEvent('Started a new game with the same deck.'); closeModal('tools-modal'); render(); scheduleSave();
  }

  function exportGame() {

  }
