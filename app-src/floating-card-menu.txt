
  // Authoritative floating card inspector + enlarged zone browsers.
  (() => {
    let forgeInspectorAnchor = null;

    function installFloatingInspectorStyle() {
      let style = document.getElementById('forge-floating-ui-style');
      if (!style) {
        style = document.createElement('style');
        style.id = 'forge-floating-ui-style';
        document.head.appendChild(style);
      }
      style.textContent = `

          #zone-modal .zone-card-grid{grid-template-columns:repeat(auto-fill,minmax(155px,1fr))!important;gap:12px!important}

    }

    function positionFloatingInspector(anchorRect) {
      const modal = document.querySelector('#inspector .inspector-modal');
      const backdrop = document.getElementById('inspector');
      if (!modal || !backdrop || backdrop.hidden) return;

      const gap = 16;
      const margin = 10;
      const rect = modal.getBoundingClientRect();
      const anchor = anchorRect || forgeInspectorAnchor?.rect || {left:innerWidth/2,right:innerWidth/2,top:70,bottom:250};

      let left = anchor.right + gap;
      if (left + rect.width > innerWidth - margin) left = anchor.left - rect.width - gap;
      if (left < margin) left = Math.max(margin, innerWidth - rect.width - margin);

      let top = anchor.top;
      if (top + rect.height > innerHeight - margin) top = innerHeight - rect.height - margin;
      top = Math.max(margin, top);

      modal.style.left = `${Math.round(left)}px`;
      modal.style.top = `${Math.round(top)}px`;
    }

    installFloatingInspectorStyle();

    window.addEventListener('contextmenu', e => {
      const el = e.target.closest?.('.remote-card');
      const id = el?.dataset.cardId;
      if (!id || !card(id)) return;
      forgeInspectorAnchor = {id,rect:el.getBoundingClientRect(),time:Date.now()};
    }, true);

    const baseOpenInspector = openInspector;
    openInspector = function(id) {
      if (!id || !card(id)) return;
      baseOpenInspector(id);
      installFloatingInspectorStyle();

      if (!forgeInspectorAnchor || forgeInspectorAnchor.id !== id || Date.now()-forgeInspectorAnchor.time > 5000) {
        const nodes = [...document.querySelectorAll('.remote-card')];
        const el = nodes.find(node => node.dataset.cardId === id && (() => {
          const r = node.getBoundingClientRect();
          return r.width>0 && r.height>0 && r.bottom>0 && r.right>0 && r.top<innerHeight && r.left<innerWidth;
        })());
        if (el) forgeInspectorAnchor = {id,rect:el.getBoundingClientRect(),time:Date.now()};
      }

      requestAnimationFrame(() => positionFloatingInspector(forgeInspectorAnchor?.rect));
    };

    window.addEventListener('resize', () => {
      if (!document.getElementById('inspector')?.hidden) positionFloatingInspector(forgeInspectorAnchor?.rect);
    });
  })();
