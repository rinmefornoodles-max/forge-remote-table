  function qolCloseDialog() { $('#qol-dialog').hidden = true; }

  function qolOpenDialog({eyebrow='TABLE ACTION', title='Action', html='', actions=[]}) {
    qolInstallUi();
    $('#qol-dialog-eyebrow').textContent = eyebrow;
    $('#qol-dialog-title').textContent = title;
    $('#qol-dialog-body').innerHTML = html;
    const footer = $('#qol-dialog-actions');
    footer.innerHTML = '';
    for (const action of actions) {
      const b = document.createElement('button');
      b.type = 'button';
      b.className = action.className || 'btn';
      b.textContent = action.label;
      b.addEventListener('click', () => action.onClick?.());
      footer.appendChild(b);
    }
    $('#qol-dialog').hidden = false;
    requestAnimationFrame(() => $('#qol-dialog-body input, #qol-dialog-body textarea, #qol-dialog-actions button')?.focus());
  }

  function qolExileFromGraveyard(amount='all') {
    const total = state.zones.graveyard.length;
    if (!total) return toast('Graveyard is empty.');
    const n = amount === 'all' ? total : clamp(Number(amount)||0, 0, total);
    if (!n) return;
    checkpoint();
    const ids = state.zones.graveyard.splice(Math.max(0,total-n), n);
    state.zones.exile.push(...ids);
    logEvent(n === total ? 'Exiled the entire graveyard.' : `Exiled ${n} card${n===1?'':'s'} from the graveyard.`);
    render(); scheduleSave();
    toast(n === total ? 'Graveyard exiled.' : `Exiled ${n} from graveyard.`);
  }

  function qolRequiredBottomCount() {
    qolEnsureState();
    return Math.max(0, Number(state.mulligansTaken)||0 - (state.mulliganFreeFirst ? 1 : 0));
  }

  function qolShuffleAndDrawSeven() {
    for (const id of [...state.zones.hand]) { removeFromZones(id); state.zones.library.push(id); }
    const a = state.zones.library;
    for (let i=a.length-1;i>0;i--) { const j=Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]]; }
    while (state.zones.hand.length < 7 && state.zones.library.length) state.zones.hand.push(state.zones.library.shift());
  }

  function qolTakeMulligan() {
    qolEnsureState();
    checkpoint();
    qolShuffleAndDrawSeven();
    state.mulligansTaken += 1;
    state.openingHandKept = false;
    qolMulliganSelected = [];
    logEvent(`Took mulligan ${state.mulligansTaken}${state.mulliganFreeFirst&&state.mulligansTaken===1?' (free multiplayer mulligan)':''}.`);
    render(); scheduleSave(); qolRenderMulligan();
  }

  function qolFinishKeep() {
    qolEnsureState();
    const required = qolRequiredBottomCount();
    if (!required) {
      checkpoint(); state.openingHandKept = true; logEvent('Kept opening hand.'); scheduleSave(); $('#qol-mulligan-modal').hidden = true; toast('Opening hand kept.'); return;
    }
    qolMulliganSelected = [];
    qolRenderMulligan('bottom');
  }

  function qolBottomSelected() {
    const required = qolRequiredBottomCount();
    if (qolMulliganSelected.length !== required) return;
    checkpoint();
    for (const id of qolMulliganSelected) {
      const i = state.zones.hand.indexOf(id);
      if (i >= 0) state.zones.hand.splice(i,1);
      state.zones.library.push(id);
    }
    state.openingHandKept = true;
    logEvent(`Kept opening hand and put ${required} card${required===1?'':'s'} on the bottom.`);
    qolMulliganSelected = [];
    render(); scheduleSave(); $('#qol-mulligan-modal').hidden = true; toast('Opening hand kept.');
  }

  function qolRenderMulligan(mode='main') {
    qolInstallUi(); qolEnsureState();
    const body = $('#qol-mulligan-body');
    const required = qolRequiredBottomCount();
    if (mode === 'bottom') {
      const cards = state.zones.hand.map(id => {
        const c = card(id); const selectedIndex = qolMulliganSelected.indexOf(id); const selected = selectedIndex >= 0;
        const image = c ? currentCardImage(c) : '';
        return `<button class="qol-hand-card${selected?' selected':''}" type="button" data-qol-bottom-id="${escapeHtml(id)}">${image?`<img src="${escapeHtml(image)}" alt="${escapeHtml(c.name)}">`:''}<strong>${escapeHtml(c?.name||'Card')}</strong>${selected?`<span class="qol-select-badge">${selectedIndex+1}</span>`:''}</button>`;
      }).join('');
      body.innerHTML = `<div class="qol-rule-box"><strong>Keep this hand:</strong> choose exactly ${required} card${required===1?'':'s'} to put on the bottom of your library. Under the London mulligan, you choose these only after deciding to keep.</div><div class="qol-mulligan-status"><span>Need to bottom <strong>${required}</strong></span><span>Selected <strong id="qol-selected-count">${qolMulliganSelected.length}</strong></span></div><div class="qol-hand-list">${cards}</div><div class="qol-dialog-actions"><button id="qol-bottom-back" class="btn ghost" type="button">Back</button><button id="qol-bottom-confirm" class="btn primary" type="button" ${qolMulliganSelected.length===required?'':'disabled'}>Put selected on bottom</button></div>`;
      $$('[data-qol-bottom-id]', body).forEach(b => b.addEventListener('click', () => {
        const id = b.dataset.qolBottomId; const i=qolMulliganSelected.indexOf(id);
        if(i>=0) qolMulliganSelected.splice(i,1); else if(qolMulliganSelected.length<required) qolMulliganSelected.push(id);
        qolRenderMulligan('bottom');
      }));
      $('#qol-bottom-back').addEventListener('click',()=>qolRenderMulligan());
      $('#qol-bottom-confirm').addEventListener('click',qolBottomSelected);
    } else {
      body.innerHTML = `<div class="qol-rule-box"><strong>London mulligan:</strong> shuffle your hand into your library and draw 7 again. When you keep, put a number of cards from your hand on the bottom equal to the mulligans that count. <br><br><strong>Commander note:</strong> in a multiplayer game, the first mulligan is free. In a normal two-player game it is not; you can switch that option below if this remote game is 1v1.</div><label class="qol-check"><input id="qol-free-mulligan" type="checkbox" ${state.mulliganFreeFirst?'checked':''}><span>Use multiplayer Commander free first mulligan</span></label><div class="qol-mulligan-status"><span>Mulligans taken <strong>${state.mulligansTaken}</strong></span><span>If you keep now, bottom <strong>${required}</strong></span><span>Hand <strong>${state.zones.hand.length}</strong></span></div>${state.openingHandKept?'<p class="qol-dialog-copy">This opening hand was already marked kept. The helper stays available because the tabletop does not enforce game timing.</p>':''}<div class="qol-dialog-actions"><button id="qol-reset-mulligan" class="btn ghost" type="button">Reset helper</button><button id="qol-take-mulligan" class="btn" type="button">Take mulligan</button><button id="qol-keep-hand" class="btn primary" type="button">Keep hand</button></div>`;
      $('#qol-free-mulligan').addEventListener('change', e => { checkpoint(); state.mulliganFreeFirst=e.target.checked; state.openingHandKept=false; scheduleSave(); qolRenderMulligan(); });
      $('#qol-reset-mulligan').addEventListener('click',()=>{checkpoint();state.mulligansTaken=0;state.openingHandKept=false;qolMulliganSelected=[];scheduleSave();qolRenderMulligan();});
      $('#qol-take-mulligan').addEventListener('click',qolTakeMulligan);
      $('#qol-keep-hand').addEventListener('click',qolFinishKeep);
    }
    $('#qol-mulligan-modal').hidden = false;
  }

  function qolDropTargetAt(clientX,clientY,dragEl) {
    const old=dragEl?.style.pointerEvents;
    if(dragEl) dragEl.style.pointerEvents='none';
    const found=document.elementFromPoint(clientX,clientY)?.closest?.('.drop-zone')||null;
    if(dragEl) dragEl.style.pointerEvents=old||'';
    return found;
  }

  function qolEnablePointerDrag(el,id,originZone) {
    if (!el || !['battlefield','hand','command'].includes(originZone)) return;
    el.draggable=false;
    el.style.touchAction='none';
    let pointerId=null,startX=0,startY=0,baseX=0,baseY=0,dragging=false,ghost=null,lastTarget=null;
    const cleanup=()=>{draggedId=null;document.body.classList.remove('qol-drag-mode');el.classList.remove('qol-dragging-card');lastTarget?.classList.remove('drag-over');lastTarget=null;ghost?.remove();ghost=null;};
    const onMove=e=>{
      if(pointerId!==e.pointerId)return;
      const dx=e.clientX-startX,dy=e.clientY-startY;
      if(!dragging&&Math.hypot(dx,dy)<7)return;
      if(!dragging){dragging=true;checkpoint();draggedId=id;document.body.classList.add('qol-drag-mode');el.classList.add('qol-dragging-card');if(originZone!=='battlefield'){ghost=el.cloneNode(true);ghost.className='qol-drag-ghost';ghost.removeAttribute('style');document.body.appendChild(ghost);}}
      e.preventDefault();
      if(originZone==='battlefield'){
        const c=card(id);if(c){c.x=Math.max(0,baseX+dx);c.y=Math.max(0,baseY+dy);el.style.left=`${c.x}px`;el.style.top=`${c.y}px`;}
      }else if(ghost){ghost.style.left=`${e.clientX}px`;ghost.style.top=`${e.clientY}px`;}
      const target=qolDropTargetAt(e.clientX,e.clientY,el);if(lastTarget!==target){lastTarget?.classList.remove('drag-over');target?.classList.add('drag-over');lastTarget=target;}
    };
    const onUp=e=>{
      if(pointerId!==e.pointerId)return;
      try{el.releasePointerCapture(pointerId);}catch{}
      if(!dragging){pointerId=null;return;}
      qolSuppressCardClicksUntil=Date.now()+420;
      const target=qolDropTargetAt(e.clientX,e.clientY,el);const targetZone=target?.dataset.dropZone;const origin=zoneOf(id);
      cleanup();pointerId=null;
      if(targetZone){
        if(targetZone==='battlefield'){
          const bf=$('#battlefield'),rect=bf.getBoundingClientRect();const x=Math.max(0,e.clientX-rect.left-61+bf.scrollLeft),y=Math.max(0,e.clientY-rect.top-85+bf.scrollTop);
          if(origin==='battlefield'){const c=card(id);if(c){c.x=x;c.y=y;logEvent(`${c.name}: repositioned on battlefield.`);render();scheduleSave();}}
          else moveCard(id,'battlefield',{x,y,checkpoint:false});
        }else if(targetZone==='library') moveCard(id,'library',{placement:'top',checkpoint:false});
        else moveCard(id,targetZone,{checkpoint:false});
      }else if(originZone==='battlefield'){render();scheduleSave();}
      else { history.pop(); render(); }
    };
    el.addEventListener('pointerdown',e=>{if(e.button!==0||pointerId!==null)return;pointerId=e.pointerId;startX=e.clientX;startY=e.clientY;const c=card(id);baseX=Number(c?.x)||0;baseY=Number(c?.y)||0;try{el.setPointerCapture(pointerId);}catch{}});
    el.addEventListener('pointermove',onMove);
    el.addEventListener('pointerup',onUp);
    el.addEventListener('pointercancel',onUp);

      checkpoint(); const commanderIds=[],all=[];
      for(const [id,c] of Object.entries(state.cards)){
        if(c.token||c.custom||c.outsideDeck){delete state.cards[id];continue;}
        c.tapped=false;c.faceDown=false;c.counters={};c.x=null;c.y=null;c.note='';c.faceIndex=0;delete c.ptBoost;delete c.stackCount;

      }
      state.zones={library:all,hand:[],battlefield:[],graveyard:[],exile:[],command:commanderIds};state.life=40;state.poison=0;state.energy=0;state.commanderTax=0;state.turn=1;state.damage=state.damage.map(d=>({...d,value:0}));state.mulligansTaken=0;state.openingHandKept=false;
