  // Simple directional duplicate stacking.
  // The FINAL DROP POINT decides the order:
  //   top half of matching card    -> put dragged card above it
  //   bottom half of matching card -> put dragged card underneath it
  (() => {
    const norm = c => String(c?.name || '').trim().toLocaleLowerCase();
    let drag = null;

    if (!document.getElementById('forge-stack-halfdrop-style')) {
      const style = document.createElement('style');
      style.id = 'forge-stack-halfdrop-style';
      style.textContent = `
        .forge-stack-drop-hint,.forge-stack-approach-hint,.forge-stack-hard-hint{display:none!important}
        .forge-stack-half-hint{
          position:absolute;left:5px;right:5px;height:46%;z-index:700;pointer-events:none;
          display:grid;place-items:center;border:2px solid #f1c779;border-radius:9px;
          background:rgba(14,29,23,.9);color:#fff5dc;font-size:12px;font-weight:950;
          letter-spacing:.06em;text-transform:uppercase;text-shadow:0 2px 4px #000;
          box-shadow:0 0 28px rgba(241,199,121,.42) inset,0 6px 18px rgba(0,0,0,.5)
        }
        .forge-stack-half-hint.top{top:5px}
        .forge-stack-half-hint.bottom{bottom:5px}
      `;
      document.head.appendChild(style);
    }

    function clearHint() {
      document.querySelectorAll('.forge-stack-half-hint').forEach(x => x.remove());
    }

    function showHint(targetEl, direction) {
      clearHint();
      if (!targetEl || !direction) return;
      const hint = document.createElement('span');
      hint.className = `forge-stack-half-hint ${direction === 'above' ? 'top' : 'bottom'}`;
      hint.textContent = direction === 'above' ? '↑ Put on top' : '↓ Put underneath';
      targetEl.appendChild(hint);
    }

    function matchingTargetAtPoint(id, clientX, clientY) {
      const moving = card(id);
      if (!moving) return null;
      let best = null;

      document.querySelectorAll('#battlefield .remote-card').forEach(node => {
        const targetId = node.dataset.cardId;
        if (!targetId || targetId === id) return;
        const target = card(targetId);
        if (!target || norm(target) !== norm(moving)) return;

        const r = node.getBoundingClientRect();
        if (clientX < r.left || clientX > r.right || clientY < r.top || clientY > r.bottom) return;

        const z = Number.parseInt(getComputedStyle(node).zIndex, 10);
        const zIndex = Number.isFinite(z) ? z : 0;
        const stateIndex = state.zones.battlefield.indexOf(targetId);
        const score = zIndex * 100000 + stateIndex;
        const direction = clientY < r.top + r.height / 2 ? 'above' : 'below';

        if (!best || score >= best.score) {
          best = {targetId, targetEl:node, rect:r, direction, score};
        }
      });

      return best;
    }

    function anchorId(id) {
      const c = card(id);
      return c?.autoStackAnchor && card(c.autoStackAnchor) ? c.autoStackAnchor : id;
    }

    function members(anchor) {
      return state.zones.battlefield.filter(sid => sid === anchor || card(sid)?.autoStackAnchor === anchor);
    }

    function reflow(anchor) {
      const base = card(anchor);
      if (!base || zoneOf(anchor) !== 'battlefield') return;
      const ids = members(anchor);
      const x = Number(base.x) || 0;
      const y = Number(base.y) || 0;
      ids.forEach((sid,index) => {
        const c = card(sid);
        if (!c) return;
        if (sid === anchor) delete c.autoStackAnchor;
        else c.autoStackAnchor = anchor;
        c.x = x + index * 10;
        c.y = y + index * 14;
      });
    }

    function detach(id) {
      const c = card(id);
      if (!c || zoneOf(id) !== 'battlefield') return;

      if (c.autoStackAnchor) {
        const oldAnchor = c.autoStackAnchor;
        delete c.autoStackAnchor;
        reflow(oldAnchor);
        return;
      }

      const children = state.zones.battlefield.filter(sid => card(sid)?.autoStackAnchor === id);
      if (!children.length) return;

      const x = Number(c.x) || 0;
      const y = Number(c.y) || 0;
      const nextAnchor = children[0];
      const nextCard = card(nextAnchor);
      if (!nextCard) return;

      delete nextCard.autoStackAnchor;
      nextCard.x = x;
      nextCard.y = y;
      children.slice(1).forEach(sid => {
        const child = card(sid);
        if (child) child.autoStackAnchor = nextAnchor;
      });
      reflow(nextAnchor);
    }

    function enforce(id, targetId, direction) {
      const moving = card(id);
      const target = card(targetId);
      if (!moving || !target || norm(moving) !== norm(target)) return false;
      if (zoneOf(targetId) !== 'battlefield') return false;

      // Remove the moving card from any existing stack first.
      if (zoneOf(id) === 'battlefield') detach(id);

      const targetAnchor = anchorId(targetId);
      const base = card(targetAnchor);
      if (!base) return false;

      const baseX = Number(base.x) || 0;
      const baseY = Number(base.y) || 0;
      const order = members(targetAnchor).filter(sid => sid !== id);
      const targetIndex = order.indexOf(targetId);
      if (targetIndex < 0) return false;

      removeFromZones(id);
      const insertAt = direction === 'above' ? targetIndex + 1 : targetIndex;
      order.splice(insertAt, 0, id);
      const unique = [...new Set(order)].filter(sid => !!card(sid));

      // Keep the stack contiguous at the end of battlefield order so visual z-order
      // follows the explicit bottom-to-top stack order.
      state.zones.battlefield = state.zones.battlefield.filter(sid => !unique.includes(sid));
      state.zones.battlefield.push(...unique);

      const newAnchor = unique[0];
      unique.forEach((sid,index) => {
        const c = card(sid);
        if (!c) return;
        if (sid === newAnchor) delete c.autoStackAnchor;
        else c.autoStackAnchor = newAnchor;
        c.x = baseX + index * 10;
        c.y = baseY + index * 14;
      });

      logEvent(`${moving.name}: put ${direction === 'above' ? 'on top of' : 'underneath'} matching card.`);
      render();
      scheduleSave();
      return true;
    }

    document.addEventListener('pointerdown', e => {
      if (e.button !== 0) return;
      const el = e.target.closest?.('.remote-card');
      const id = el?.dataset.cardId;
      if (!id || !card(id)) return;
      const origin = zoneOf(id);
      if (!['hand','battlefield','command'].includes(origin)) return;

      drag = {
        pointerId:e.pointerId,
        id,
        startX:e.clientX,
        startY:e.clientY,
        active:false,
        moveWholeStack:!!e.shiftKey
      };
    }, true);

    document.addEventListener('pointermove', e => {
      if (!drag || e.pointerId !== drag.pointerId) return;
      const dx = e.clientX - drag.startX;
      const dy = e.clientY - drag.startY;
      if (!drag.active && Math.hypot(dx,dy) < 7) return;
      drag.active = true;

      if (drag.moveWholeStack) {
        clearHint();
        return;
      }

      const target = matchingTargetAtPoint(drag.id, e.clientX, e.clientY);
      if (target) showHint(target.targetEl, target.direction);
      else clearHint();
    }, true);

    document.addEventListener('pointerup', e => {
      if (!drag || e.pointerId !== drag.pointerId) return;
      const current = drag;
      drag = null;
      clearHint();

      if (!current.active || current.moveWholeStack) return;

      // Capture the exact card and half under the mouse at release time, then let
      // the normal drag/drop handler finish before enforcing stack order.
      const target = matchingTargetAtPoint(current.id, e.clientX, e.clientY);
      if (!target) return;

      const {targetId, direction} = target;
      setTimeout(() => {
        if (!card(current.id) || !card(targetId)) return;
        enforce(current.id, targetId, direction);
      }, 0);
    }, true);

    document.addEventListener('pointercancel', e => {
      if (drag && e.pointerId === drag.pointerId) {
        drag = null;
        clearHint();
      }
    }, true);
  })();
