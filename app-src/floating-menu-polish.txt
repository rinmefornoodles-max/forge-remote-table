  // Directional stack hard guard.
  // Independent capture-phase tracker + post-drop enforcer. This runs after the
  // normal drag system and reapplies the intended duplicate stack order last.
  (() => {
    const norm = c => String(c?.name || '').trim().toLocaleLowerCase();
    let drag = null;

    if (!document.getElementById('forge-stack-hardguard-style')) {
      const style = document.createElement('style');
      style.id = 'forge-stack-hardguard-style';
      style.textContent = `
        .forge-stack-drop-hint,.forge-stack-approach-hint{display:none!important}
        .forge-stack-hard-hint{
          position:absolute;left:5px;right:5px;z-index:600;pointer-events:none;
          display:grid;place-items:center;height:46%;border:2px solid #f1c779;border-radius:9px;
          background:rgba(14,29,23,.90);color:#fff5dc;font-size:12px;font-weight:950;
          letter-spacing:.06em;text-transform:uppercase;text-shadow:0 2px 4px #000;
          box-shadow:0 0 28px rgba(241,199,121,.42) inset,0 6px 18px rgba(0,0,0,.5)
        }
        .forge-stack-hard-hint.top{top:5px}
        .forge-stack-hard-hint.bottom{bottom:5px}
      `;
      document.head.appendChild(style);
    }

    function clearHint() {
      document.querySelectorAll('.forge-stack-hard-hint').forEach(x => x.remove());
    }

    function showHint(targetEl, direction) {
      clearHint();
      if (!targetEl || !direction) return;
      const el = document.createElement('span');
      el.className = `forge-stack-hard-hint ${direction === 'above' ? 'top' : 'bottom'}`;
      el.textContent = direction === 'above' ? '↑ Put on top' : '↓ Put underneath';
      targetEl.appendChild(el);
    }

    function overlapArea(a,b) {
      const w = Math.max(0, Math.min(a.right,b.right) - Math.max(a.left,b.left));
      const h = Math.max(0, Math.min(a.bottom,b.bottom) - Math.max(a.top,b.top));
      return w*h;
    }

    function translatedRect(base, dx, dy) {
      return {
        left: base.left + dx,
        right: base.right + dx,
        top: base.top + dy,
        bottom: base.bottom + dy,
        width: base.width,
        height: base.height
      };
    }

    function bestMatchingTarget(id, rect) {
      const moving = card(id);
      if (!moving) return null;
      let best = null;
      document.querySelectorAll('#battlefield .remote-card').forEach(node => {
        const targetId = node.dataset.cardId;
        if (!targetId || targetId === id) return;
        const target = card(targetId);
        if (!target || norm(target) !== norm(moving)) return;
        const r = node.getBoundingClientRect();
        const area = overlapArea(rect,r);
        if (area <= 0) return;
        const cx=(rect.left+rect.right)/2, cy=(rect.top+rect.bottom)/2;
        const tx=(r.left+r.right)/2, ty=(r.top+r.bottom)/2;
        const distance=Math.hypot(cx-tx,cy-ty);
        const score=area-distance*.1;
        if (!best || score > best.score) best={targetId,targetEl:node,rect:r,score};
      });
      return best;
    }

    function anchorId(id) {
      const c=card(id);
      return c?.autoStackAnchor && card(c.autoStackAnchor) ? c.autoStackAnchor : id;
    }

    function members(anchor) {
      return state.zones.battlefield.filter(sid => sid===anchor || card(sid)?.autoStackAnchor===anchor);
    }

    function reflow(anchor) {
      const base=card(anchor);
      if (!base || zoneOf(anchor)!=='battlefield') return;
      const ids=members(anchor);
      const x=Number(base.x)||0, y=Number(base.y)||0;
      ids.forEach((sid,index) => {
        const c=card(sid);
        if (!c) return;
        if (sid===anchor) delete c.autoStackAnchor;
        else c.autoStackAnchor=anchor;
        c.x=x+index*10;
        c.y=y+index*14;
      });
    }

    function detach(id) {
      const c=card(id);
      if (!c || zoneOf(id)!=='battlefield') return;
      if (c.autoStackAnchor) {
        const old=c.autoStackAnchor;
        delete c.autoStackAnchor;
        reflow(old);
        return;
      }
      const kids=state.zones.battlefield.filter(sid => card(sid)?.autoStackAnchor===id);
      if (!kids.length) return;
      const x=Number(c.x)||0, y=Number(c.y)||0;
      const next=kids[0], nextCard=card(next);
      if (!nextCard) return;
      delete nextCard.autoStackAnchor;
      nextCard.x=x; nextCard.y=y;
      kids.slice(1).forEach(sid => { const m=card(sid); if(m) m.autoStackAnchor=next; });
      reflow(next);
    }

    function enforce(id,targetId,direction) {
      const moving=card(id), target=card(targetId);
      if (!moving || !target || norm(moving)!==norm(target)) return false;
      if (zoneOf(targetId)!=='battlefield') return false;

      if (zoneOf(id)==='battlefield') detach(id);

      const targetAnchor=anchorId(targetId);
      const base=card(targetAnchor);
      if (!base) return false;
      const baseX=Number(base.x)||0, baseY=Number(base.y)||0;
      const order=members(targetAnchor).filter(sid => sid!==id);
      const targetIndex=order.indexOf(targetId);
      if (targetIndex<0) return false;

      removeFromZones(id);
      const insertAt=direction==='above' ? targetIndex+1 : targetIndex;
      order.splice(insertAt,0,id);
      const unique=[...new Set(order)].filter(sid => !!card(sid));

      state.zones.battlefield=state.zones.battlefield.filter(sid => !unique.includes(sid));
      state.zones.battlefield.push(...unique);

      const newAnchor=unique[0];
      unique.forEach((sid,index) => {
        const c=card(sid);
        if (!c) return;
        if (sid===newAnchor) delete c.autoStackAnchor;
        else c.autoStackAnchor=newAnchor;
        c.x=baseX+index*10;
        c.y=baseY+index*14;
      });

      logEvent(`${moving.name}: put ${direction==='above'?'on top of':'underneath'} matching card.`);
      render();
      scheduleSave();
      return true;
    }

    document.addEventListener('pointerdown', e => {
      if (e.button!==0) return;
      const el=e.target.closest?.('.remote-card');
      const id=el?.dataset.cardId;
      if (!id || !card(id)) return;
      const origin=zoneOf(id);
      if (!['hand','battlefield','command'].includes(origin)) return;
      const rect=el.getBoundingClientRect();
      drag={
        pointerId:e.pointerId,id,origin,el,
        startX:e.clientX,startY:e.clientY,
        baseRect:{left:rect.left,right:rect.right,top:rect.top,bottom:rect.bottom,width:rect.width,height:rect.height},
        active:false,lastCenterY:(rect.top+rect.bottom)/2,targetId:null,direction:null
      };
    }, true);

    document.addEventListener('pointermove', e => {
      if (!drag || e.pointerId!==drag.pointerId) return;
      const dx=e.clientX-drag.startX, dy=e.clientY-drag.startY;
      if (!drag.active && Math.hypot(dx,dy)<7) return;
      drag.active=true;

      const rect=translatedRect(drag.baseRect,dx,dy);
      const candidate=bestMatchingTarget(drag.id,rect);
      const centerY=(rect.top+rect.bottom)/2;

      if (candidate) {
        if (candidate.targetId!==drag.targetId) {
          const targetCenter=(candidate.rect.top+candidate.rect.bottom)/2;
          if (drag.lastCenterY < candidate.rect.top + candidate.rect.height*.35) drag.direction='above';
          else if (drag.lastCenterY > candidate.rect.bottom - candidate.rect.height*.35) drag.direction='below';
          else if (dy>0) drag.direction='above';
          else if (dy<0) drag.direction='below';
          else drag.direction=drag.lastCenterY<=targetCenter?'above':'below';
          drag.targetId=candidate.targetId;
        }
        showHint(candidate.targetEl,drag.direction);
      } else {
        drag.targetId=null;
        drag.direction=null;
        clearHint();
      }
      drag.lastCenterY=centerY;
    }, true);

    document.addEventListener('pointerup', e => {
      if (!drag || e.pointerId!==drag.pointerId) return;
      const current=drag;
      drag=null;
      clearHint();
      if (!current.active || !current.targetId || !current.direction) return;

      const dx=e.clientX-current.startX, dy=e.clientY-current.startY;
      const rect=translatedRect(current.baseRect,dx,dy);
      const candidate=bestMatchingTarget(current.id,rect);
      const targetId=candidate?.targetId || current.targetId;
      const direction=current.direction;

      setTimeout(() => {
        if (!card(current.id) || !card(targetId)) return;
        enforce(current.id,targetId,direction);
      },0);
    }, true);

    document.addEventListener('pointercancel', e => {
      if (drag && e.pointerId===drag.pointerId) { drag=null; clearHint(); }
    }, true);
  })();
