    function ensureDiceState() {
      if (!Array.isArray(state.dice)) state.dice = [];
      state.dice = state.dice.filter(d => d && d.id && ['plus','minus','d20'].includes(d.kind));
      for (const die of state.dice) {
        die.value = clamp(Number(die.value) || 1, 1, die.kind === 'd20' ? 20 : 6);
        die.x = Math.max(0, Number(die.x) || 0);
        die.y = Math.max(0, Number(die.y) || 0);
        die.rx = Number.isFinite(Number(die.rx)) ? Number(die.rx) : -14;
        die.ry = Number.isFinite(Number(die.ry)) ? Number(die.ry) : 18;
        die.rz = Number.isFinite(Number(die.rz)) ? Number(die.rz) : 0;
      }
      return state.dice;
    }

    function counterLabel(kind, value) {
      const sign = kind === 'minus' ? '−' : '+';
      return `${sign}${value}/${sign}${value}`;
    }

    function maxFor(die) { return die.kind === 'd20' ? 20 : 6; }
    function wrapValue(value, max) { return ((Number(value) - 1 + max * 20) % max) + 1; }

    function snapOrientation(die) {
      if (die.kind === 'd20') {
        const v = die.value;
        die.rx = -34 + ((v * 47) % 118);
        die.ry = 18 + ((v * 73) % 360);
        die.rz = -24 + ((v * 29) % 48);
        return;
      }
      const map = {
        1:[-14,18,0],
        2:[-102,18,0],
        3:[-14,-72,0],
        4:[-14,108,0],
        5:[76,18,0],
        6:[-14,198,0]
      };
      [die.rx, die.ry, die.rz] = map[die.value] || map[1];
    }

    function setDieValue(die, value, shouldSnap=true) {
      die.value = wrapValue(value, maxFor(die));
      if (shouldSnap) snapOrientation(die);
    }

    function dieById(id) {
      ensureDiceState();
      return state.dice.find(d => d.id === id) || null;
    }

    function updateDieDom(el, die) {
      el.style.left = `${Math.max(0, Number(die.x)||0)}px`;
      el.style.top = `${Math.max(0, Number(die.y)||0)}px`;
      const solid = el.querySelector('.forge-die-solid');
      if (solid) solid.style.transform = `rotateX(${die.rx}deg) rotateY(${die.ry}deg) rotateZ(${die.rz}deg)`;
      const readout = el.querySelector('.forge-d20-readout');
      if (readout) readout.textContent = String(die.value);
      if (die.kind !== 'd20') {
        for (const face of el.querySelectorAll('.forge-die-face[data-face-value]')) {
          const value = Number(face.dataset.faceValue) || 1;
          face.textContent = counterLabel(die.kind, value);
        }
      }
      el.title = die.kind === 'd20'
        ? `D20 — ${die.value}. Drag to move. Shift-drag to turn. Right-click for menu.`
        : `${die.kind === 'minus' ? '−/−' : '+/+'} counter die — ${counterLabel(die.kind, die.value)}. Drag to move. Shift-drag to turn. Right-click for menu.`;
    }

    function makeD20Solid(die) {
      const solid = document.createElement('div');
      solid.className = 'forge-die-solid forge-d20-solid';
      for (let i=0;i<20;i++) {
        const face = document.createElement('span');
        face.className = 'forge-d20-face';
        face.textContent = String(i+1);
        const ring = Math.floor(i/5);
        const slot = i % 5;
        const pitch = [-58,-20,20,58][ring];
        const yaw = slot * 72 + (ring % 2 ? 36 : 0);
        const twist = ring % 2 ? 180 : 0;
        face.style.transform = `rotateY(${yaw}deg) rotateX(${pitch}deg) rotateZ(${twist}deg) translateZ(30px)`;
        solid.appendChild(face);
      }
      return solid;
    }

    function makeCubeSolid(die) {
      const solid = document.createElement('div');
      solid.className = 'forge-die-solid';
      for (const [side,value] of CUBE_FACES) {
        const face = document.createElement('span');
        face.className = `forge-die-face ${side}`;
        face.dataset.faceValue = String(value);
        face.textContent = counterLabel(die.kind, value);
        solid.appendChild(face);
      }
      return solid;
    }

    function deleteDie(die) {
      ensureDiceState();
      checkpoint();
      state.dice = state.dice.filter(d => d.id !== die.id);
      closeDieMenu();
      render();
      scheduleSave();
      toast('Die removed.');
    }

    function ensureDieMenu() {
      let menu = document.getElementById('forge-die-context-menu');
      if (menu) return menu;
      menu = document.createElement('div');
      menu.id = 'forge-die-context-menu';
      menu.hidden = true;
      document.body.appendChild(menu);

      menu.addEventListener('pointerdown', e => e.stopPropagation());
      menu.addEventListener('click', e => {
        const die = dieById(openDieMenuId);
        if (!die) return closeDieMenu();

        const valueButton = e.target.closest?.('[data-die-menu-value]');
        if (valueButton) {
          const value = Number(valueButton.dataset.dieMenuValue);
          if (Number.isFinite(value)) {
            checkpoint();
            setDieValue(die, value, true);
            renderDice();
            scheduleSave();
            openDieMenu(die, Number(menu.dataset.anchorX)||0, Number(menu.dataset.anchorY)||0);
          }
          return;
        }

        const action = e.target.closest?.('[data-die-menu-action]')?.dataset.dieMenuAction;
        if (action === 'turn') {
          checkpoint();
          setDieValue(die, die.value + 1, true);
          renderDice();
          scheduleSave();
          openDieMenu(die, Number(menu.dataset.anchorX)||0, Number(menu.dataset.anchorY)||0);
        } else if (action === 'rotate') {
          checkpoint();
          die.rz = (Number(die.rz)||0) + 45;
          renderDice();
          scheduleSave();
        } else if (action === 'delete') {
          deleteDie(die);
        }
      });
      return menu;
    }

    function closeDieMenu() {
      openDieMenuId = null;
      const menu = document.getElementById('forge-die-context-menu');
      if (menu) menu.hidden = true;
    }

    function openDieMenu(die, clientX, clientY) {
      const menu = ensureDieMenu();
      openDieMenuId = die.id;
      const max = maxFor(die);
      const title = die.kind === 'd20' ? 'D20' : (die.kind === 'minus' ? '−1/−1 counter die' : '+1/+1 counter die');
      const current = die.kind === 'd20' ? String(die.value) : counterLabel(die.kind, die.value);
      const values = [];
      for (let value=1; value<=max; value++) {
        const label = die.kind === 'd20' ? String(value) : counterLabel(die.kind, value);
        values.push(`<button class="forge-die-menu-value${value === die.value ? ' active' : ''}" type="button" data-die-menu-value="${value}">${label}</button>`);
      }
      menu.innerHTML = `<div class="forge-die-menu-head"><div><strong>${title}</strong><span>Current face: ${current}</span></div></div><div class="forge-die-menu-values${die.kind === 'd20' ? ' d20' : ''}">${values.join('')}</div><div class="forge-die-menu-actions"><button class="btn tiny" type="button" data-die-menu-action="turn">Turn next face</button><button class="btn tiny ghost" type="button" data-die-menu-action="rotate">Rotate 45°</button><button class="btn tiny danger" type="button" data-die-menu-action="delete" style="grid-column:1/-1">Remove die</button></div>`;
      menu.hidden = false;
      menu.dataset.anchorX = String(clientX);
      menu.dataset.anchorY = String(clientY);
      menu.style.left = '0px';
      menu.style.top = '0px';
      const rect = menu.getBoundingClientRect();
      const margin = 8;
      const left = clamp(clientX + 8, margin, Math.max(margin, innerWidth - rect.width - margin));
      const top = clamp(clientY + 8, margin, Math.max(margin, innerHeight - rect.height - margin));
      menu.style.left = `${Math.round(left)}px`;
      menu.style.top = `${Math.round(top)}px`;
    }

    function wireDieInteraction(el, die) {
      let gesture = null;

      el.addEventListener('contextmenu', e => {
        e.preventDefault();
        e.stopPropagation();
        openDieMenu(die, e.clientX, e.clientY);
      });

      el.addEventListener('pointerdown', e => {
        if (e.button !== 0) return;
        closeDieMenu();
        e.preventDefault();
        e.stopPropagation();
        const mode = e.shiftKey ? 'turn' : 'move';
        gesture = {
          pointerId:e.pointerId, mode,
          startClientX:e.clientX, startClientY:e.clientY,
          startX:Number(die.x)||0, startY:Number(die.y)||0,
          startRx:Number(die.rx)||0, startRy:Number(die.ry)||0, startRz:Number(die.rz)||0,
          changed:false, checkpointed:false
        };
        el.classList.toggle('is-turning', mode === 'turn');
        try { el.setPointerCapture(e.pointerId); } catch {}
      });

      el.addEventListener('pointermove', e => {
        if (!gesture || gesture.pointerId !== e.pointerId) return;
        const dx = e.clientX - gesture.startClientX;
        const dy = e.clientY - gesture.startClientY;
        if (!gesture.changed && Math.hypot(dx,dy) < 3) return;
        if (!gesture.checkpointed) {
          checkpoint();
          gesture.checkpointed = true;
        }
        gesture.changed = true;
        e.preventDefault();
        e.stopPropagation();

        if (gesture.mode === 'move') {
          const bf = $('#battlefield');
          const maxX = Math.max(0, (bf?.scrollWidth || bf?.clientWidth || 0) - 36);
          const maxY = Math.max(0, (bf?.scrollHeight || bf?.clientHeight || 0) - 36);
          die.x = clamp(gesture.startX + dx, 0, maxX || 99999);
          die.y = clamp(gesture.startY + dy, 0, maxY || 99999);
        } else {
          die.rx = gesture.startRx - dy * 1.25;
          die.ry = gesture.startRy + dx * 1.25;
          die.rz = gesture.startRz + dx * .18;
        }
        updateDieDom(el, die);
      });

      const finish = e => {
        if (!gesture || (e.pointerId != null && gesture.pointerId !== e.pointerId)) return;
        const changed = gesture.changed;
        gesture = null;
        el.classList.remove('is-turning');
        try { if (e.pointerId != null) el.releasePointerCapture(e.pointerId); } catch {}
        if (changed) scheduleSave();
      };
      el.addEventListener('pointerup', finish);
      el.addEventListener('pointercancel', finish);
    }

    function makeDieElement(die) {
      const el = document.createElement('div');
      el.className = `forge-table-die forge-die-${die.kind}`;
      el.dataset.dieId = die.id;
      el.tabIndex = 0;

      const scene = document.createElement('div');
      scene.className = 'forge-die-scene';
      scene.appendChild(die.kind === 'd20' ? makeD20Solid(die) : makeCubeSolid(die));
      if (die.kind === 'd20') {
        const readout = document.createElement('span');
        readout.className = 'forge-d20-readout';
        readout.textContent = String(die.value);
        scene.appendChild(readout);
      }
      el.appendChild(scene);

      updateDieDom(el, die);
      wireDieInteraction(el, die);
      return el;
    }

    function renderDice() {
      if (!state.started) return;
      ensureDiceState();
      const bf = $('#battlefield');
      if (!bf) return;
      bf.querySelectorAll('.forge-table-die').forEach(el => el.remove());
      for (const die of state.dice) bf.appendChild(makeDieElement(die));
    }

    function createDie(kind) {
      if (!['plus','minus','d20'].includes(kind)) return;
      ensureDiceState();
      const bf = $('#battlefield');
      const offset = state.dice.length % 7;
      const x = Math.max(8, (bf?.scrollLeft || 0) + Math.min(180, (bf?.clientWidth || 500) * .42) + offset * 12);
      const y = Math.max(8, (bf?.scrollTop || 0) + Math.min(150, (bf?.clientHeight || 500) * .28) + offset * 10);
      const die = { id:uid(), kind, value:1, x, y, rx:-14, ry:18, rz:0 };
      snapOrientation(die);
      checkpoint();
      state.dice.push(die);
      renderDice();
      scheduleSave();
      toast(kind === 'd20' ? 'D20 placed on the table.' : `${kind === 'minus' ? '−/−' : '+/+'} counter die placed.`);
    }

    function installDiceUi() {
      installDiceStyle();
      ensureDieMenu();

      const actions = document.querySelector('.battlefield-actions');
      if (actions && !document.getElementById('forge-dice-btn')) {
        const button = document.createElement('button');
        button.id = 'forge-dice-btn';
        button.className = 'btn tiny';
        button.type = 'button';
        button.textContent = '+ Dice';
        const tokenButton = document.getElementById('token-btn');
        if (tokenButton?.parentElement === actions) tokenButton.insertAdjacentElement('afterend', button);
        else actions.appendChild(button);
      }

      if (!document.getElementById('forge-dice-modal')) {
        const modal = document.createElement('div');
        modal.id = 'forge-dice-modal';
        modal.className = 'modal-backdrop';
        modal.hidden = true;
        modal.innerHTML = `<section class="modal small-modal" role="dialog" aria-modal="true" aria-labelledby="forge-dice-title"><button class="modal-close" type="button" data-close-dice aria-label="Close">×</button><div class="eyebrow">TABLETOP OBJECTS</div><h2 id="forge-dice-title">Create a die</h2><p class="muted-block">Place physical-style dice anywhere on the battlefield, including directly over cards.</p><div class="forge-dice-picks"><button class="btn forge-dice-pick" type="button" data-create-die="plus"><strong>+1/+1 counter die</strong><span>Green resin-style D6</span></button><button class="btn forge-dice-pick" type="button" data-create-die="minus"><strong>−1/−1 counter die</strong><span>Red resin-style D6</span></button><button class="btn forge-dice-pick primary" type="button" data-create-die="d20"><strong>D20</strong><span>Faceted tabletop die</span></button></div><p class="forge-dice-help"><strong>Drag</strong> to move a die. <strong>Shift + drag</strong> to physically turn it. <strong>Right-click</strong> the die for its face/value menu or to remove it. Dice always stay above cards and stacks.</p></section>`;
        document.body.appendChild(modal);
        modal.querySelector('[data-close-dice]')?.addEventListener('click', () => closeModal('forge-dice-modal'));
        modal.addEventListener('mousedown', e => { if (e.target === modal) closeModal('forge-dice-modal'); });
        modal.addEventListener('click', e => {
          const pick = e.target.closest?.('[data-create-die]');
          if (!pick) return;
          createDie(pick.dataset.createDie);
        });
      }

      document.getElementById('forge-dice-btn')?.addEventListener('click', () => openModal('forge-dice-modal'));

      const grid = document.querySelector('#tools-modal .tool-grid');
      if (grid && !document.getElementById('tool-dice')) {
        const tile = document.createElement('button');
        tile.id = 'tool-dice';
        tile.className = 'tool-tile';
        tile.type = 'button';
        tile.innerHTML = '<strong>Create tabletop dice</strong><span>Place +/+, −/−, or D20 dice over cards.</span>';
        const tokenTile = document.getElementById('tool-token');
        if (tokenTile?.parentElement === grid) tokenTile.insertAdjacentElement('afterend', tile);
        else grid.appendChild(tile);
        tile.addEventListener('click', () => { closeModal('tools-modal'); openModal('forge-dice-modal'); });
      }

      document.addEventListener('pointerdown', e => {
        const menu = document.getElementById('forge-die-context-menu');
        if (!menu || menu.hidden) return;
        if (menu.contains(e.target)) return;
        if (e.target.closest?.('.forge-table-die')) return;
        closeDieMenu();
      }, true);
      document.addEventListener('keydown', e => { if (e.key === 'Escape') closeDieMenu(); });
      window.addEventListener('resize', closeDieMenu);
    }

    const forgeDiceBaseRender = render;
    render = function() {
      forgeDiceBaseRender();
      renderDice();
    };

    installDiceUi();
    ensureDiceState();
    requestAnimationFrame(renderDice);
