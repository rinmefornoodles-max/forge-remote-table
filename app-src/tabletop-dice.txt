    function ensureDiceState() {
      if (!Array.isArray(state.dice)) state.dice = [];
      state.dice = state.dice.filter(d => d && d.id && ['plus','minus','d20'].includes(d.kind));
      for (const d of state.dice) {
        d.value = clamp(Number(d.value)||1,1,d.kind==='d20'?20:6);
        d.x = Math.max(0,Number(d.x)||0); d.y = Math.max(0,Number(d.y)||0);
        d.rx = Number.isFinite(Number(d.rx)) ? Number(d.rx) : -14;
        d.ry = Number.isFinite(Number(d.ry)) ? Number(d.ry) : 18;
        d.rz = Number.isFinite(Number(d.rz)) ? Number(d.rz) : 0;
      }
      return state.dice;
    }

    const maxFor = d => d.kind==='d20'?20:6;
    const wrapValue = (v,max) => ((Number(v)-1+max*20)%max)+1;
    function counterLabel(kind,value){const s=kind==='minus'?'−':'+';return `${s}${value}/${s}${value}`;}
    function snapOrientation(d){
      if(d.kind==='d20'){const v=d.value;d.rx=-18+((v*37)%54);d.ry=12+((v*61)%360);d.rz=-15+((v*23)%30);return;}
      const map={1:[-14,18,0],2:[-102,18,0],3:[-14,-72,0],4:[-14,108,0],5:[76,18,0],6:[-14,198,0]};[d.rx,d.ry,d.rz]=map[d.value]||map[1];
    }
    function setDieValue(d,value,snap=true){d.value=wrapValue(value,maxFor(d));if(snap)snapOrientation(d);}
    function dieById(id){return ensureDiceState().find(d=>d.id===id)||null;}

    function counterMark(kind,value){const s=kind==='minus'?'−':'+';return `<span class="forge-counter-mark"><span class="a">${s}${value}</span><i class="slash"></i><span class="b">${s}${value}</span></span>`;}
    function makeCubeSolid(die){
      const solid=document.createElement('div');solid.className='forge-die-solid';
      for(const [side,value] of CUBE_FACES){const face=document.createElement('span');face.className=`forge-die-face ${side}`;face.dataset.faceValue=String(value);face.innerHTML=counterMark(die.kind,value);solid.appendChild(face);}return solid;
    }
    function makeD20Solid(die){
      const solid=document.createElement('div');solid.className='forge-die-solid forge-d20-solid';
      const shell=document.createElement('div');shell.className='forge-d20-shell';
      shell.innerHTML=`<svg viewBox="0 0 100 100" aria-hidden="true"><polygon class="f1" points="50,2 20,12 34,37 66,37 80,12"/><polygon class="f2" points="20,12 3,36 34,37"/><polygon class="f3" points="80,12 97,36 66,37"/><polygon class="f4" points="3,36 8,69 34,37"/><polygon class="f1" points="97,36 92,69 66,37"/><polygon class="f3" points="34,37 50,57 66,37"/><polygon class="f2" points="8,69 28,93 50,57 34,37"/><polygon class="f4" points="92,69 72,93 50,57 66,37"/><polygon class="f1" points="28,93 50,99 50,57"/><polygon class="f3" points="72,93 50,99 50,57"/></svg>`;
      const n=document.createElement('span');n.className='forge-d20-number';n.textContent=String(die.value);solid.append(shell,n);return solid;
    }
    function updateDieDom(el,die){
      el.style.left=`${Math.max(0,Number(die.x)||0)}px`;el.style.top=`${Math.max(0,Number(die.y)||0)}px`;
      const solid=el.querySelector('.forge-die-solid');if(solid)solid.style.transform=`rotateX(${die.rx}deg) rotateY(${die.ry}deg) rotateZ(${die.rz}deg)`;
      const n=el.querySelector('.forge-d20-number');if(n)n.textContent=String(die.value);
      el.title=die.kind==='d20'?`D20 — ${die.value}. Drag to move. Shift-drag to turn. Right-click for menu.`:`${counterLabel(die.kind,die.value)} counter die. Drag to move. Shift-drag to turn. Right-click for menu.`;
    }

    function ensureDieMenu(){
      let m=document.getElementById('forge-die-context-menu');if(m)return m;m=document.createElement('div');m.id='forge-die-context-menu';m.hidden=true;document.body.appendChild(m);
      m.addEventListener('pointerdown',e=>e.stopPropagation());
      m.addEventListener('click',e=>{const d=dieById(openDieMenuId);if(!d)return closeDieMenu();const vb=e.target.closest?.('[data-die-menu-value]');if(vb){checkpoint();setDieValue(d,Number(vb.dataset.dieMenuValue),true);renderDice();scheduleSave();openDieMenu(d,Number(m.dataset.anchorX)||0,Number(m.dataset.anchorY)||0);return;}const a=e.target.closest?.('[data-die-menu-action]')?.dataset.dieMenuAction;if(a==='turn'){checkpoint();setDieValue(d,d.value+1,true);renderDice();scheduleSave();openDieMenu(d,Number(m.dataset.anchorX)||0,Number(m.dataset.anchorY)||0);}else if(a==='rotate'){checkpoint();d.rz=(Number(d.rz)||0)+45;renderDice();scheduleSave();}else if(a==='delete'){checkpoint();state.dice=state.dice.filter(x=>x.id!==d.id);closeDieMenu();render();scheduleSave();toast('Die removed.');}});
      return m;
    }
    function closeDieMenu(){openDieMenuId=null;const m=document.getElementById('forge-die-context-menu');if(m)m.hidden=true;}
    function openDieMenu(die,x,y){
      const m=ensureDieMenu();openDieMenuId=die.id;m.dataset.anchorX=String(x);m.dataset.anchorY=String(y);const max=maxFor(die);const vals=Array.from({length:max},(_,i)=>i+1);
      m.innerHTML=`<div class="forge-die-menu-head"><strong>${die.kind==='d20'?'D20':(die.kind==='minus'?'−1/−1 counter die':'+1/+1 counter die')}</strong><span>Choose the face/value</span></div><div class="forge-die-menu-values ${die.kind==='d20'?'d20':''}">${vals.map(v=>`<button class="forge-die-menu-value ${v===die.value?'active':''}" data-die-menu-value="${v}">${die.kind==='d20'?v:counterLabel(die.kind,v)}</button>`).join('')}</div><div class="forge-die-menu-actions"><button class="btn tiny" data-die-menu-action="turn">Next face</button><button class="btn tiny" data-die-menu-action="rotate">Rotate 45°</button><button class="btn tiny danger" style="grid-column:1/-1" data-die-menu-action="delete">Remove die</button></div>`;
      m.hidden=false;requestAnimationFrame(()=>{const r=m.getBoundingClientRect();m.style.left=`${Math.max(8,Math.min(innerWidth-r.width-8,x))}px`;m.style.top=`${Math.max(8,Math.min(innerHeight-r.height-8,y))}px`;});
    }

    function wireDie(el,die){
      let g=null;
      el.addEventListener('contextmenu',e=>{e.preventDefault();e.stopImmediatePropagation();openDieMenu(die,e.clientX+8,e.clientY+8);});
      el.addEventListener('pointerdown',e=>{if(e.button!==0)return;e.preventDefault();e.stopPropagation();closeDieMenu();g={id:e.pointerId,sx:e.clientX,sy:e.clientY,x:die.x,y:die.y,v:die.value,changed:false,cp:false,turn:e.shiftKey};el.classList.toggle('is-turning',g.turn);try{el.setPointerCapture(e.pointerId)}catch{}});
      el.addEventListener('pointermove',e=>{if(!g||g.id!==e.pointerId)return;const dx=e.clientX-g.sx,dy=e.clientY-g.sy;if(!g.changed&&Math.hypot(dx,dy)<3)return;if(!g.cp){checkpoint();g.cp=true}g.changed=true;e.preventDefault();e.stopPropagation();if(g.turn){setDieValue(die,g.v+Math.trunc((dx-dy)/30),true);}else{const bf=$('#battlefield');die.x=clamp(g.x+dx,0,Math.max(0,(bf?.scrollWidth||99999)-40));die.y=clamp(g.y+dy,0,Math.max(0,(bf?.scrollHeight||99999)-40));}updateDieDom(el,die);});
      const finish=e=>{if(!g||g.id!==e.pointerId)return;const changed=g.changed;g=null;el.classList.remove('is-turning');try{el.releasePointerCapture(e.pointerId)}catch{}if(changed)scheduleSave();};el.addEventListener('pointerup',finish);el.addEventListener('pointercancel',finish);
    }
    function makeDieElement(die){const el=document.createElement('div');el.className=`forge-table-die forge-die-${die.kind}`;el.dataset.dieId=die.id;const scene=document.createElement('div');scene.className='forge-die-scene';scene.appendChild(die.kind==='d20'?makeD20Solid(die):makeCubeSolid(die));el.appendChild(scene);updateDieDom(el,die);wireDie(el,die);return el;}
    function renderDice(){if(!state.started)return;ensureDiceState();const bf=$('#battlefield');if(!bf)return;bf.querySelectorAll('.forge-table-die').forEach(x=>x.remove());for(const d of state.dice)bf.appendChild(makeDieElement(d));}
    function createDie(kind){if(!['plus','minus','d20'].includes(kind))return;ensureDiceState();const bf=$('#battlefield'),o=state.dice.length%7;const d={id:uid(),kind,value:1,x:Math.max(8,(bf?.scrollLeft||0)+120+o*12),y:Math.max(8,(bf?.scrollTop||0)+95+o*10),rx:-14,ry:18,rz:0};snapOrientation(d);checkpoint();state.dice.push(d);renderDice();scheduleSave();toast(kind==='d20'?'D20 placed on the table.':'Counter die placed on the table.');}

    function installDiceUi(){
      installDiceStyle();ensureDieMenu();const actions=document.querySelector('.battlefield-actions');if(actions&&!document.getElementById('forge-dice-btn')){const b=document.createElement('button');b.id='forge-dice-btn';b.className='btn tiny';b.type='button';b.textContent='+ Dice';const t=document.getElementById('token-btn');if(t?.parentElement===actions)t.insertAdjacentElement('afterend',b);else actions.appendChild(b);}
      if(!document.getElementById('forge-dice-modal')){const modal=document.createElement('div');modal.id='forge-dice-modal';modal.className='modal-backdrop';modal.hidden=true;modal.innerHTML=`<section class="modal small-modal" role="dialog" aria-modal="true"><button class="modal-close" type="button" data-close-dice>×</button><div class="eyebrow">TABLETOP OBJECTS</div><h2>Create a die</h2><p class="muted-block">Physical-style dice that stay above cards and stacks.</p><div class="forge-dice-picks"><button class="btn forge-dice-pick" data-create-die="plus"><strong>+1/+1 die</strong><span>Black marbled resin / gold engraving</span></button><button class="btn forge-dice-pick" data-create-die="minus"><strong>−1/−1 die</strong><span>Ivory marbled resin / gold engraving</span></button><button class="btn forge-dice-pick primary" data-create-die="d20"><strong>D20</strong><span>Faceted 20-sided counter die</span></button></div><p class="forge-dice-help"><strong>Drag</strong> to move. <strong>Shift + drag</strong> to turn. <strong>Right-click</strong> for values, rotation, or removal.</p></section>`;document.body.appendChild(modal);modal.querySelector('[data-close-dice]').addEventListener('click',()=>closeModal('forge-dice-modal'));modal.addEventListener('mousedown',e=>{if(e.target===modal)closeModal('forge-dice-modal')});modal.addEventListener('click',e=>{const p=e.target.closest?.('[data-create-die]');if(p)createDie(p.dataset.createDie)});}
      document.getElementById('forge-dice-btn')?.addEventListener('click',()=>openModal('forge-dice-modal'));
      const grid=document.querySelector('#tools-modal .tool-grid');if(grid&&!document.getElementById('tool-dice')){const tile=document.createElement('button');tile.id='tool-dice';tile.className='tool-tile';tile.type='button';tile.innerHTML='<strong>Create tabletop dice</strong><span>Place +/+, −/−, or D20 dice over cards.</span>';grid.appendChild(tile);tile.addEventListener('click',()=>{closeModal('tools-modal');openModal('forge-dice-modal')});}
      document.addEventListener('pointerdown',e=>{const m=document.getElementById('forge-die-context-menu');if(m&&!m.hidden&&!m.contains(e.target)&&!e.target.closest?.('.forge-table-die'))closeDieMenu();},true);
    }

    const baseRender=render;render=function(){baseRender();renderDice();};installDiceUi();ensureDiceState();requestAnimationFrame(renderDice);
  })();
