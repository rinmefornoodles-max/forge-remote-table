
  function showApp(){ $('#start-screen').hidden=true; $('#app').hidden=false; window.scrollTo(0,0); }

  function continueSaved(){
    try{const raw=JSON.parse(localStorage.getItem(STORAGE_KEY)||'null');if(!raw?.started)return;state=normalizeLoadedState(raw);history=[];future=[];showApp();render();}
    catch{toast('Saved table could not be opened.','error');}
  }

  async function loadPreconIndex() {
    const status=$('#precon-status'); const list=$('#precon-list');
    setStatus(status,'Loading Commander Forge official-precon catalog…'); list.innerHTML='';
    try{
      const res=await fetch(PRECON_INDEX_URL,{cache:'no-store'}); if(!res.ok)throw new Error(`Catalog unavailable (${res.status})`);
      const payload=await res.json();
      preconRows = payload.decks || payload.items || payload.data || (Array.isArray(payload)?payload:[]);
      if(!Array.isArray(preconRows)) preconRows=[];
      setStatus(status,`${preconRows.length} official deck entries available.`,'success'); renderPreconList();
    }catch(err){setStatus(status,'The live precon catalog could not be reached. You can still paste any decklist.','error');}
  }

  function renderPreconList() {
    const q=$('#precon-search').value.trim().toLocaleLowerCase(); const root=$('#precon-list');root.innerHTML='';
    const rows=preconRows.filter(row=>!q||JSON.stringify([row.name,row.commanderNames,row.releaseDate,row.code]).toLocaleLowerCase().includes(q)).slice(0,160);
    for(const row of rows){
      const el=document.createElement('div');el.className='precon-item';
      const commanders=Array.isArray(row.commanderNames)?row.commanderNames.join(' / '):(row.commander||'');
      el.innerHTML=`<div><strong>${escapeHtml(row.name||'Commander deck')}</strong><span>${escapeHtml(commanders)}${row.releaseDate?` · ${escapeHtml(row.releaseDate)}`:''}</span></div><button class="btn tiny primary" type="button">Use deck</button>`;
      el.querySelector('button').addEventListener('click',()=>loadPrecon(row)); root.appendChild(el);
    }
    if(!rows.length)root.innerHTML='<p class="muted-block">No matching precons.</p>';
  }

  async function loadPrecon(row) {
    const status=$('#precon-status'); setStatus(status,`Loading ${row.name||'deck'}…`);
    const file=row.localFile || row.file || row.path || row.outputFile || row.slug;
    if(!file){setStatus(status,'This catalog row did not include a deck file. Paste a decklist instead.','error');return;}
    try{
      const url=/^https?:/i.test(file)?file:`${PRECON_BASE_URL}${file}`;
      const res=await fetch(url,{cache:'no-store'});if(!res.ok)throw new Error(`Deck file unavailable (${res.status})`);const payload=await res.json();
      const deck=payload?.data || payload;
      const entries=deck.entries||deck.cards||[];
      const lines=[];
      const commanders=deck.commanderNames||row.commanderNames||[];
      if(commanders.length){lines.push('Commander',...commanders.map(name=>`1 ${name}`),'');}
      lines.push('Deck');
      for(const e of entries){
        const name=typeof e==='string'?e:e.name; const qty=typeof e==='string'?1:(e.count||e.quantity||1); if(!name)continue;
        // Avoid double-counting commander rows if the precon file also includes them in entries.
        const isCommander=commanders.some(c=>c.toLocaleLowerCase()===String(name).toLocaleLowerCase());
        if(isCommander) continue;
        lines.push(`${qty} ${name}`);
      }
      $('#deck-name').value=deck.name||row.name||'Official precon';
      $('#commander-name').value=commanders[0]||'';
      $('#decklist-input').value=lines.join('\n');
      closeModal('precon-modal'); setStatus($('#setup-status'),`${deck.name||row.name} loaded into the deck box. Click “Load deck & draw 7”.`,'success');
    }catch(err){setStatus(status,`Could not load that precon: ${err.message||err}`,'error');}
  }

  function wireDropZones() {
    $$('.drop-zone').forEach(el=>{
      el.addEventListener('dragover',e=>{if(!draggedId)return;e.preventDefault();e.dataTransfer.dropEffect='move';el.classList.add('drag-over');});
      el.addEventListener('dragleave',e=>{if(!el.contains(e.relatedTarget))el.classList.remove('drag-over');});
      el.addEventListener('drop',e=>{
        e.preventDefault();el.classList.remove('drag-over');const id=draggedId||e.dataTransfer.getData('text/plain');if(!id||!card(id))return;
        let zone=el.dataset.dropZone;
        if(zone==='battlefield'){
          const rect=$('#battlefield').getBoundingClientRect();const c=card(id);const x=e.clientX-rect.left-61+$('#battlefield').scrollLeft;const y=e.clientY-rect.top-85+$('#battlefield').scrollTop;
          if(zoneOf(id)==='battlefield'){
            checkpoint();c.x=Math.max(0,x);c.y=Math.max(0,y);render();scheduleSave();
          } else moveCard(id,'battlefield',{x:Math.max(0,x),y:Math.max(0,y)});
        } else if(zone==='library') moveCard(id,'library',{placement:'top'}); else moveCard(id,zone);
      });
    });
  }
