

  // High-resolution Scryfall card images.
  // Prefer Scryfall PNG/large images for newly loaded cards and transparently
  // upgrade older saved-game normal/small URLs at display time.
  (() => {
    function forgeHighResCardUrl(url) {
      if (!url) return '';
      const value = String(url);
      if (!/cards\.scryfall\.io/i.test(value)) return value;
      return value
        .replace('/small/', '/large/')
        .replace('/normal/', '/large/');
    }

    const forgeBaseImageForScryfall = imageForScryfall;
    imageForScryfall = function(data, faceIndex=0) {
      if (!data) return '';
      const uris = data.image_uris || data.card_faces?.[faceIndex]?.image_uris || data.card_faces?.[0]?.image_uris;
      if (uris) return uris.png || uris.large || uris.normal || uris.small || forgeBaseImageForScryfall(data, faceIndex) || '';
      return forgeBaseImageForScryfall(data, faceIndex) || '';
    };

    const forgeBaseCardTemplate = cardTemplate;
    cardTemplate = function(data, fallbackName) {
      const template = forgeBaseCardTemplate(data, fallbackName);
      const topUris = data?.image_uris;
      if (topUris) template.image = topUris.png || topUris.large || topUris.normal || topUris.small || template.image || '';

      if (Array.isArray(data?.card_faces) && Array.isArray(template.faces)) {
        template.faces = template.faces.map((face, index) => {
          const uris = data.card_faces[index]?.image_uris;
          if (!uris) return face;
          return {
            ...face,
            image: uris.png || uris.large || uris.normal || uris.small || face.image || ''
          };
        });
      }
      return template;
    };

    currentCardImage = function(c) {
      if (!c) return '';
      const raw = c.faces?.length
        ? (c.faces[c.faceIndex]?.image || c.image || '')
        : (c.image || '');
      return forgeHighResCardUrl(raw);
    };

    let style = document.getElementById('forge-hires-card-style');
    if (!style) {
      style = document.createElement('style');
      style.id = 'forge-hires-card-style';
      document.head.appendChild(style);
    }
    style.textContent = `
      .remote-card img,
      .zone-tile .remote-card img {
        image-rendering:auto!important;
        object-fit:cover;
        width:100%;
        height:100%;
      }

      /* Large inspector previews must show the WHOLE card, not crop it. */
      #inspector .inspector-image,
      .inspector-image {
        image-rendering:auto!important;
        display:block!important;
        width:100%!important;
        max-width:100%!important;
        height:auto!important;
        max-height:none!important;
        aspect-ratio:auto!important;
        object-fit:contain!important;
        object-position:center center!important;
        transform:none!important;
        clip-path:none!important;
      }
    `;
  })();

  // Universal top/bottom stacking.
  // Any card may be stacked onto any battlefield card. The half of the target
  // under the mouse at release decides the exact order.
  (() => {
    let forgeUniversalStackDrag = null;

    function forgeStackAnchor(id) {
      const c = card(id);
      return c?.autoStackAnchor && card(c.autoStackAnchor) ? c.autoStackAnchor : id;
    }

    function forgeStackMembersAny(anchorId) {
      return state.zones.battlefield.filter(id => id === anchorId || card(id)?.autoStackAnchor === anchorId);
    }

    function forgeReflowAnyStack(anchorId) {
      const anchor = card(anchorId);
      if (!anchor || zoneOf(anchorId) !== 'battlefield') return;
      const ids = forgeStackMembersAny(anchorId);
      const baseX = Number(anchor.x) || 0;
      const baseY = Number(anchor.y) || 0;
      ids.forEach((id,index) => {
        const c = card(id);
        if (!c) return;
        if (id === anchorId) delete c.autoStackAnchor;
        else c.autoStackAnchor = anchorId;
        c.x = baseX + index * 10;
        c.y = baseY + index * 14;
      });
    }

    function forgeDetachAnyCard(id) {
      const c = card(id);
      if (!c || zoneOf(id) !== 'battlefield') return;

      if (c.autoStackAnchor) {
        const oldAnchor = c.autoStackAnchor;
        delete c.autoStackAnchor;
        forgeReflowAnyStack(oldAnchor);
        return;
      }

      const children = state.zones.battlefield.filter(otherId => card(otherId)?.autoStackAnchor === id);
      if (!children.length) return;

      const baseX = Number(c.x) || 0;
      const baseY = Number(c.y) || 0;
      const nextAnchorId = children[0];
      const nextAnchor = card(nextAnchorId);
      if (!nextAnchor) return;

      delete nextAnchor.autoStackAnchor;
      nextAnchor.x = baseX;
      nextAnchor.y = baseY;
      for (const childId of children.slice(1)) {
        const child = card(childId);
        if (child) child.autoStackAnchor = nextAnchorId;
      }
      forgeReflowAnyStack(nextAnchorId);
    }

    function forgeUniversalTargetAt(x,y,sourceId) {
      const seen = new Set();
      for (const node of document.elementsFromPoint(x,y)) {
        const cardEl = node.closest?.('#battlefield .remote-card');
        if (!cardEl || seen.has(cardEl)) continue;
        seen.add(cardEl);
        const targetId = cardEl.dataset.cardId;
        if (!targetId || targetId === sourceId || zoneOf(targetId) !== 'battlefield') continue;
        return { id: targetId, el: cardEl, rect: cardEl.getBoundingClientRect() };
      }
      return null;
    }

    function forgeApplyUniversalStack(sourceId,targetId,direction) {
      const source = card(sourceId);
      const target = card(targetId);
      if (!source || !target || zoneOf(targetId) !== 'battlefield') return false;

      if (zoneOf(sourceId) === 'battlefield') forgeDetachAnyCard(sourceId);

      const targetAnchorId = forgeStackAnchor(targetId);
      const targetAnchor = card(targetAnchorId);
      if (!targetAnchor) return false;

      const baseX = Number(targetAnchor.x) || 0;
      const baseY = Number(targetAnchor.y) || 0;
      const order = forgeStackMembersAny(targetAnchorId).filter(id => id !== sourceId);
      const targetIndex = order.indexOf(targetId);
      if (targetIndex < 0) return false;

      removeFromZones(sourceId);
      const insertAt = direction === 'above' ? targetIndex + 1 : targetIndex;
      order.splice(insertAt,0,sourceId);

      const uniqueOrder = [...new Set(order)].filter(id => !!card(id));
      state.zones.battlefield = state.zones.battlefield.filter(id => !uniqueOrder.includes(id));
      state.zones.battlefield.push(...uniqueOrder);

      const newAnchorId = uniqueOrder[0];
      uniqueOrder.forEach((id,index) => {
        const c = card(id);
        if (!c) return;
        if (id === newAnchorId) delete c.autoStackAnchor;
        else c.autoStackAnchor = newAnchorId;
        c.x = baseX + index * 10;
        c.y = baseY + index * 14;
      });

      logEvent(`${source.name || 'Card'}: stacked ${direction === 'above' ? 'above' : 'below'} ${target.name || 'card'}.`);
      render();
      scheduleSave();
      return true;
    }

    document.addEventListener('pointerdown', event => {
      if (event.button !== 0) return;
      const el = event.target.closest?.('.remote-card');
      const id = el?.dataset.cardId;
      if (!id || !card(id)) return;
      const origin = zoneOf(id);
      if (!['hand','battlefield','command'].includes(origin)) return;
      forgeUniversalStackDrag = {
        pointerId: event.pointerId,
        id,
        startX: event.clientX,
        startY: event.clientY,
        active: false
      };
    }, true);

    document.addEventListener('pointermove', event => {
      const drag = forgeUniversalStackDrag;
      if (!drag || event.pointerId !== drag.pointerId) return;
      if (!drag.active && Math.hypot(event.clientX-drag.startX,event.clientY-drag.startY) >= 7) drag.active = true;
    }, true);

    document.addEventListener('pointerup', event => {
      const drag = forgeUniversalStackDrag;
      if (!drag || event.pointerId !== drag.pointerId) return;
      forgeUniversalStackDrag = null;
      if (!drag.active) return;

      const target = forgeUniversalTargetAt(event.clientX,event.clientY,drag.id);
      if (!target) return;
      const direction = event.clientY < target.rect.top + target.rect.height / 2 ? 'above' : 'below';
      const sourceId = drag.id;
      const targetId = target.id;

      // Run after the normal drag/drop and the older duplicate-only helper.
      setTimeout(() => {
        if (!card(sourceId) || !card(targetId) || zoneOf(targetId) !== 'battlefield') return;
        forgeApplyUniversalStack(sourceId,targetId,direction);
      }, 25);
    }, true);

    document.addEventListener('pointercancel', event => {
      if (forgeUniversalStackDrag && event.pointerId === forgeUniversalStackDrag.pointerId) forgeUniversalStackDrag = null;
    }, true);
  })();

  // Smooth progressive stack browsing.
  // A stack opens like a restrained accordion. The top card peels first; when
  // the pointer settles over the next revealed card, the fan opens one layer
  // farther. Positions are visual-only and collapse smoothly back to state.
  (() => {
    const MIN_STEP = 30;
    const MAX_STEP = 46;
    const TARGET_SPREAD = 185;
    const NEXT_CARD_DWELL = 95;
    const COLLAPSE_MS = 220;

    let activeAnchorId = null;
    let peelDepth = 0;
    let pendingId = null;
    let pendingTimer = null;
    let collapseToken = 0;

    if (!document.getElementById('forge-stack-peel-style')) {
      const style = document.createElement('style');
      style.id = 'forge-stack-peel-style';
      style.textContent = `
        #battlefield .remote-card.forge-stack-browse-card {
          transition:top .22s cubic-bezier(.22,.78,.22,1),filter .16s ease,box-shadow .16s ease!important;
          will-change:top;
        }
        #battlefield .remote-card.forge-stack-browse-card:hover {
          z-index:500!important;
          filter:brightness(1.06);
          box-shadow:0 12px 28px rgba(0,0,0,.5),0 0 0 2px rgba(208,164,95,.38);
        }
        #battlefield .remote-card.forge-stack-peeled {
          filter:brightness(1.035);
        }
        @media (prefers-reduced-motion:reduce) {
          #battlefield .remote-card.forge-stack-browse-card { transition:none!important; }
        }
      `;
      document.head.appendChild(style);
    }

    function stackAnchorFor(id) {
      const c = card(id);
      if (!c || zoneOf(id) !== 'battlefield') return null;
      return c.autoStackAnchor && card(c.autoStackAnchor) ? c.autoStackAnchor : id;
    }

    function stackMembersFor(anchorId) {
      if (!anchorId) return [];
      return state.zones.battlefield.filter(id => id === anchorId || card(id)?.autoStackAnchor === anchorId);
    }

    function nodeFor(id) {
      return document.querySelector(`#battlefield .remote-card[data-card-id="${CSS.escape(id)}"]`);
    }

    function stepFor(ids) {
      const gaps = Math.max(1, ids.length - 1);
      return Math.max(MIN_STEP, Math.min(MAX_STEP, TARGET_SPREAD / gaps));
    }

    function cancelPending() {
      if (pendingTimer) clearTimeout(pendingTimer);
      pendingTimer = null;
      pendingId = null;
    }

    function applyPeel() {
      if (!activeAnchorId) return;
      const ids = stackMembersFor(activeAnchorId);
      if (ids.length < 2) return collapsePeel();

      const topDown = [...ids].reverse();
      const step = stepFor(ids);
      ids.forEach(id => {
        const c = card(id);
        const node = nodeFor(id);
        if (!c || !node) return;
        node.classList.add('forge-stack-browse-card');
        const topIndex = topDown.indexOf(id);
        const layer = topIndex >= 0 && topIndex < peelDepth ? peelDepth - topIndex : 0;
        node.classList.toggle('forge-stack-peeled', layer > 0);
        node.style.top = `${(Number(c.y) || 0) + layer * step}px`;
      });
    }

    function collapsePeel(immediate=false) {
      cancelPending();
      if (!activeAnchorId) return;

      const oldAnchor = activeAnchorId;
      const ids = stackMembersFor(oldAnchor);
      const token = ++collapseToken;
      activeAnchorId = null;
      peelDepth = 0;

      ids.forEach(id => {
        const c = card(id);
        const node = nodeFor(id);
        if (!node) return;
        node.classList.add('forge-stack-browse-card');
        node.classList.remove('forge-stack-peeled');
        if (c) node.style.top = `${Number(c.y) || 0}px`;
      });

      const cleanup = () => {
        if (token !== collapseToken || activeAnchorId) return;
        ids.forEach(id => nodeFor(id)?.classList.remove('forge-stack-browse-card','forge-stack-peeled'));
      };

      if (immediate) cleanup();
      else setTimeout(cleanup, COLLAPSE_MS + 30);
    }

    function stackCardAt(x,y) {
      const seen = new Set();
      for (const hit of document.elementsFromPoint(x,y)) {
        const el = hit.closest?.('#battlefield .remote-card');
        if (!el || seen.has(el)) continue;
        seen.add(el);
        const id = el.dataset.cardId;
        if (!id || zoneOf(id) !== 'battlefield') continue;
        const anchorId = stackAnchorFor(id);
        const members = stackMembersFor(anchorId);
        if (members.length < 2) continue;
        return { id, anchorId, members };
      }
      return null;
    }

    function pointInsideActiveStack(x,y) {
      if (!activeAnchorId) return false;
      const rects = stackMembersFor(activeAnchorId)
        .map(id => nodeFor(id)?.getBoundingClientRect())
        .filter(Boolean);
      if (!rects.length) return false;
      const left = Math.min(...rects.map(r => r.left)) - 18;
      const right = Math.max(...rects.map(r => r.right)) + 18;
      const top = Math.min(...rects.map(r => r.top)) - 18;
      const bottom = Math.max(...rects.map(r => r.bottom)) + 18;
      return x >= left && x <= right && y >= top && y <= bottom;
    }

    function queueNextLayer(info) {
      if (!activeAnchorId || info.anchorId !== activeAnchorId) return;
      const ids = stackMembersFor(activeAnchorId);
      const topDown = [...ids].reverse();
      const targetDepth = topDown.indexOf(info.id) + 1;
      if (targetDepth <= peelDepth || targetDepth < 1) {
        cancelPending();
        return;
      }
      if (pendingId === info.id) return;

      cancelPending();
      pendingId = info.id;
      pendingTimer = setTimeout(() => {
        if (!activeAnchorId || pendingId !== info.id) return;
        const currentIds = stackMembersFor(activeAnchorId);
        const currentTopDown = [...currentIds].reverse();
        const currentDepth = currentTopDown.indexOf(info.id) + 1;
        if (currentDepth > peelDepth) {
          peelDepth = currentDepth;
          applyPeel();
        }
        pendingTimer = null;
        pendingId = null;
      }, NEXT_CARD_DWELL);
    }

    document.addEventListener('pointermove', event => {
      if (event.pointerType === 'touch') return;
      if (draggedId || document.body.classList.contains('qol-drag-mode')) {
        collapsePeel(true);
        return;
      }

      const info = stackCardAt(event.clientX,event.clientY);
      if (!info) {
        cancelPending();
        if (!pointInsideActiveStack(event.clientX,event.clientY)) collapsePeel();
        return;
      }

      if (activeAnchorId !== info.anchorId) {
        collapsePeel(true);
        collapseToken++;
        activeAnchorId = info.anchorId;
        peelDepth = 1;
        applyPeel();
        return;
      }

      queueNextLayer(info);
    }, true);

    // Real interactions use canonical card coordinates, never temporary browse positions.
    document.addEventListener('pointerdown', () => collapsePeel(true), true);
    document.addEventListener('contextmenu', () => collapsePeel(true), true);
    window.addEventListener('blur', () => collapsePeel(true));

    const battlefield = document.getElementById('battlefield');
    battlefield?.addEventListener('pointerleave', event => {
      if (!event.buttons) collapsePeel();
    });
  })();
