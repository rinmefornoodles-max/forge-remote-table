  function luckClearTimers() {
    if (luckAnimationTimer) clearTimeout(luckAnimationTimer);
    if (luckNumberTimer) clearInterval(luckNumberTimer);
    luckAnimationTimer = null;
    luckNumberTimer = null;
  }

  function luckClose() {
    luckClearTimers();
    const modal = $('#luck-modal');
    if (modal) modal.hidden = true;
  }

  function luckSparks() {
    const stage = $('#luck-stage');
    if (!stage) return;
    const sparks = document.createElement('div');
    sparks.className = 'luck-sparks';
    for (let i=0;i<14;i++) {
      const s = document.createElement('span');
      s.className = 'luck-spark';
      s.style.setProperty('--angle', `${i * (360/14)}deg`);
      s.style.setProperty('--delay', `${(i%4)*24}ms`);
      sparks.appendChild(s);
    }
    stage.appendChild(sparks);
    setTimeout(() => sparks.remove(), 900);
  }

  function luckShowModal(kind, title) {
    luckInstallUi();
    luckClearTimers();
    $('#luck-kicker').textContent = kind;
    $('#luck-title').textContent = title;
    $('#luck-stage').innerHTML = '<div class="luck-shadow"></div>';
    $('#luck-result').textContent = '';
    $('#luck-sub').textContent = '';
    $('#luck-actions').innerHTML = '';
    $('#luck-modal').hidden = false;
  }

  function luckAction(label, className, handler) {
    const b = document.createElement('button');
    b.type = 'button';
    b.className = className || 'btn';
    b.textContent = label;
    b.addEventListener('click', handler);
    $('#luck-actions').appendChild(b);
  }

  function luckShowDicePicker() {
    qolOpenDialog({
      eyebrow:'DICE', title:'Choose a die',
      html:'<p class="qol-dialog-copy">Pick the die you want to roll. The result will land visibly on the animated die.</p>',
      actions:[
        {label:'Cancel',className:'btn ghost',onClick:qolCloseDialog},
        {label:'Roll D6',className:'btn',onClick:()=>{qolCloseDialog();luckRollDie(6);}},
        {label:'Roll D20',className:'btn primary',onClick:()=>{qolCloseDialog();luckRollDie(20);}}
      ]
    });
  }

  function luckRollDie(sides) {
    sides = sides === 6 ? 6 : 20;
    const result = 1 + Math.floor(Math.random() * sides);
    luckShowModal('DICE ROLL', `Rolling D${sides}`);
    $('#random-result').textContent = `Rolling d${sides}…`;
    $('#luck-sub').textContent = 'The die is in motion…';
    const stage = $('#luck-stage');

    if (sides === 6) {
      const orientation = {
        1:[720,1080,720], 6:[720,1260,720], 3:[720,990,720],
        4:[720,1170,720], 2:[630,1080,720], 5:[810,1080,720]
      }[result];
      const cube = document.createElement('div');
      cube.className = 'luck-cube-wrap luck-object';
      cube.innerHTML = `<div class="luck-cube" style="--rx:${orientation[0]}deg;--ry:${orientation[1]}deg;--rz:${orientation[2]}deg"><div class="luck-cube-face front">1</div><div class="luck-cube-face back">6</div><div class="luck-cube-face right">3</div><div class="luck-cube-face left">4</div><div class="luck-cube-face top">2</div><div class="luck-cube-face bottom">5</div></div>`;
      stage.appendChild(cube);
    } else {
      const d20 = document.createElement('div');
      d20.className = 'luck-d20 luck-object';
      d20.innerHTML = `<span class="luck-d20-number">1</span>`;
      stage.appendChild(d20);
      const number = d20.querySelector('.luck-d20-number');
      luckNumberTimer = setInterval(() => { number.textContent = String(1 + Math.floor(Math.random()*20)); }, 72);
    }

    const reduced = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches;
    luckAnimationTimer = setTimeout(() => {
      if (luckNumberTimer) { clearInterval(luckNumberTimer); luckNumberTimer = null; }
      const number = $('.luck-d20-number'); if (number) number.textContent = String(result);
      $('.luck-object')?.classList.add('luck-landing');
      $('#luck-result').textContent = `D${sides}: ${result}`;
      $('#luck-sub').textContent = result === sides ? `Natural ${result}!` : 'Result locked in.';
      $('#random-result').textContent = `d${sides}: ${result}`;
      logEvent(`Rolled d${sides}: ${result}.`);
      luckSparks();
      luckAction(`Roll D${sides} again`,'btn primary',()=>luckRollDie(sides));
      luckAction('Other die','btn ghost',()=>{luckClose();luckShowDicePicker();});
      luckAnimationTimer = null;
    }, reduced ? 30 : 1580);
  }

  function luckFlipCoin() {
    const result = Math.random() < .5 ? 'Heads' : 'Tails';
    const end = result === 'Heads' ? 1800 : 1980;
    luckShowModal('COIN FLIP', 'Flipping coin');
    $('#random-result').textContent = 'Flipping…';
    $('#luck-sub').textContent = 'The coin is in the air…';
    const stage = $('#luck-stage');
    const scene = document.createElement('div');
    scene.className = 'luck-coin-scene luck-object';
    scene.innerHTML = `<div class="luck-coin" style="--coin-end:${end}deg"><div class="luck-coin-face front"><div class="luck-coin-emblem">HEADS</div></div><div class="luck-coin-face back"><div class="luck-coin-emblem">TAILS</div></div></div>`;
    stage.appendChild(scene);
    const reduced = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches;
    luckAnimationTimer = setTimeout(() => {
      scene.classList.add('luck-landing');
      $('#luck-result').textContent = result;
      $('#luck-sub').textContent = `The coin landed on ${result}.`;
      $('#random-result').textContent = result;
      logEvent(`Coin flip: ${result}.`);
      luckSparks();
      luckAction('Flip again','btn primary',luckFlipCoin);
      luckAction('Done','btn ghost',luckClose);
      luckAnimationTimer = null;
    }, reduced ? 30 : 1660);
  }

  // Capture-phase handlers intentionally run before the legacy card click/randomizer handlers.
  document.addEventListener('click', e => {
    const roll = e.target.closest?.('#roll-btn');
    if (roll) {
      e.preventDefault(); e.stopImmediatePropagation();
      luckShowDicePicker();
      return;
    }
    const coin = e.target.closest?.('#coin-btn');
    if (coin) {
      e.preventDefault(); e.stopImmediatePropagation();
      luckFlipCoin();
      return;
    }
    const cardEl = e.target.closest?.('.remote-card');
    if (cardEl) {
      // Keep single-left-click from opening the inspector. Dragging and dblclick still work.
      e.stopImmediatePropagation();
    }
  }, true);

  document.addEventListener('contextmenu', e => {
    const cardEl = e.target.closest?.('.remote-card');
    if (!cardEl) return;
    const id = cardEl.dataset.cardId;
    if (!id || !card(id)) return;
    e.preventDefault();
    e.stopImmediatePropagation();
    selectedId = id;
    openInspector(id);
  }, true);

  luckInstallUi();
