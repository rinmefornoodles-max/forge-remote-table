(() => {

  const STORAGE_KEY = 'forge-remote-table-state-v1';
  const DECK_CACHE_KEY = 'forge-remote-table-last-deck-v1';

  const ZONES = ['library','hand','battlefield','graveyard','exile','command'];

  const $ = (sel, root=document) => root.querySelector(sel);
  const $$ = (sel, root=document) => [...root.querySelectorAll(sel)];
  const deepCopy = (obj) => JSON.parse(JSON.stringify(obj));
  const uid = () => (crypto?.randomUUID?.() || `card-${Date.now()}-${Math.random().toString(36).slice(2)}`);
  const clamp = (n,min,max) => Math.max(min,Math.min(max,n));
  const escapeHtml = (value='') => String(value).replace(/[&<>'"]/g, ch => ({'&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#39;','"':'&quot;'}[ch]));

  let state = freshState();
  let history = [];
  let future = [];
  let selectedId = null;
  let draggedId = null;
  let zoneModalZone = null;
  let zoneModalLimit = null;
  let saveTimer = null;
  let preconRows = [];

  function freshState() {
    return {
      version: 1,
      started: false,
      deckName: 'Remote Commander',
      commanderName: '',
      life: 40,
      poison: 0,
      energy: 0,
      commanderTax: 0,
      turn: 1,
      notes: '',
      damage: [],
      cards: {},
      zones: { library: [], hand: [], battlefield: [], graveyard: [], exile: [], command: [] },
      eventLog: []
    };
  }

  function normalizeLoadedState(raw) {
    const next = Object.assign(freshState(), raw || {});
    next.cards = raw?.cards && typeof raw.cards === 'object' ? raw.cards : {};
    next.zones = Object.assign(freshState().zones, raw?.zones || {});
    for (const zone of ZONES) {
      if (!Array.isArray(next.zones[zone])) next.zones[zone] = [];
      next.zones[zone] = next.zones[zone].filter(id => next.cards[id]);
    }
    next.damage = Array.isArray(next.damage) ? next.damage : [];
    return next;
  }

  function card(id) { return state.cards[id] || null; }
  function zoneOf(id) { return ZONES.find(z => state.zones[z].includes(id)) || null; }

  function toast(message, kind='') {
    const root = $('#toast-root');
    const node = document.createElement('div');
    node.className = `toast ${kind}`.trim();
    node.textContent = message;
    root.appendChild(node);
    setTimeout(() => node.remove(), 3300);
  }

  function setStatus(el, message='', kind='') {
    el.textContent = message;
    el.className = `status ${kind}`.trim();
  }

  function logEvent(message) {
    state.eventLog.unshift({ at: new Date().toISOString(), message });
    state.eventLog = state.eventLog.slice(0, 80);
  }

  function checkpoint() {
    history.push(deepCopy(state));
    if (history.length > 50) history.shift();
    future = [];
  }

  function undo() {
    if (!history.length) return toast('Nothing to undo.');
    future.push(deepCopy(state));
    state = history.pop();
    selectedId = null;
    render();
    scheduleSave();
  }

  function redo() {
    if (!future.length) return toast('Nothing to redo.');
    history.push(deepCopy(state));
    state = future.pop();
    selectedId = null;
    render();
    scheduleSave();
  }

  function scheduleSave() {
    clearTimeout(saveTimer);
    saveTimer = setTimeout(() => {
      try { localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); } catch (err) { console.warn(err); }
      updateContinueButton();
    }, 180);
  }

  function updateContinueButton() {
    let has = false;
    try { has = Boolean(JSON.parse(localStorage.getItem(STORAGE_KEY) || 'null')?.started); } catch {}
    $('#continue-game').hidden = !has;
  }

  function openModal(id) {
    const el = document.getElementById(id);
    if (!el) return;
    el.hidden = false;
    requestAnimationFrame(() => el.querySelector('input,textarea,button:not(.modal-close)')?.focus());
  }
  function closeModal(id) {
    const el = document.getElementById(id);
    if (el) el.hidden = true;
    if (id === 'inspector') selectedId = null;
  }
  function closeAllModals() { $$('.modal-backdrop').forEach(el => el.hidden = true); selectedId = null; }

  function parseDecklist(text, explicitCommander='') {
    const lines = String(text || '').replace(/\r/g,'').split('\n');
    const entries = [];
    const quantityByName = new Map();
    let section = '';
    let autoCommander = '';

    const categoryWords = /^(commander|commanders|companion|creatures?|lands?|artifacts?|enchantments?|instants?|sorceries?|planeswalkers?|battles?|other|spells?|sideboard|maybeboard|tokens?|deck|mainboard)(?:\s*[-:(].*)?$/i;

    for (let raw of lines) {
      let line = raw.trim();
      if (!line || /^\/\//.test(line) || /^#/.test(line)) continue;
      line = line.replace(/^[-•]\s*/, '').trim();
      if (!line) continue;

      const cleanHeader = line.replace(/^\[|\]$/g,'').trim();
      if (categoryWords.test(cleanHeader) && !/^\d/.test(cleanHeader)) {
        section = cleanHeader.toLowerCase().startsWith('commander') ? 'commander' : cleanHeader.toLowerCase();
        continue;
      }

      const match = line.match(/^(?:(\d+)\s*[xX]?\s+)?(.+?)\s*$/);
      if (!match) continue;
      let qty = Number(match[1] || 1);
      let name = match[2].trim();

      // Common deck export suffixes: "(SET) 123", foil markers, arena collector IDs.
      name = name.replace(/\s+\([A-Z0-9]{2,8}\)\s+[A-Za-z0-9-]+(?:\s+\*[^*]+\*)?$/i, '');
      name = name.replace(/\s+\[[A-Z0-9]{2,8}\]$/i, '');
      name = name.replace(/\s+\*[^*]+\*$/i, '');
      name = name.trim();
      if (!name || categoryWords.test(name)) continue;

      qty = clamp(Number.isFinite(qty) ? qty : 1, 1, 200);
      const key = name.toLocaleLowerCase();
      const previous = quantityByName.get(key);
      quantityByName.set(key, { name: previous?.name || name, quantity: (previous?.quantity || 0) + qty });
      if (section === 'commander' && !autoCommander) autoCommander = name;
